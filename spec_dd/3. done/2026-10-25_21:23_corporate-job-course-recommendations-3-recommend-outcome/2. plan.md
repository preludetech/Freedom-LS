```python
def remove_duplicates(apps: StateApps, schema_editor: BaseDatabaseSchemaEditor) -> None:
    """Keep the oldest row per (site, user, course) and delete the rest.

    Every row so far was made by staff and carries nothing but its timestamp,
    so the extra rows lose no data. The oldest stays so the learner's dashboard
    order, newest first, does not move.
    """
    RecommendedCourse = apps.get_model("freedom_ls_course_recommendations", "RecommendedCourse")
    seen: set[tuple[int, int, UUID]] = set()
    extra: list[UUID] = []
    for pk, site_id, user_id, course_id in (
        RecommendedCourse.objects.order_by("created_at", "id")
        .values_list("id", "site_id", "user_id", "course_id")
        .iterator()
    ):
        key = (site_id, user_id, course_id)
        if key in seen:
            extra.append(pk)
        else:
            seen.add(key)
    RecommendedCourse.objects.filter(id__in=extra).delete()


class Migration(migrations.Migration):
    dependencies = [("freedom_ls_course_recommendations", "0001_initial")]
    operations = [migrations.RunPython(remove_duplicates, migrations.RunPython.noop)]
```

## Slice 2: every registration path clears a recommendation the user no longer needs

**Behaviour:** when a user becomes registered for a course by any path that saves a row, their
recommendation for that course is deleted, whoever made it. Self-registration keeps clearing,
now through the same receiver.

### Shared conditions and the set-based sibling (`freedom_ls/learner_management/queries.py`)

```python
#: An individual registration grants access while it and its Learner are both active.
LEARNER_REGISTRATION_GRANTS_ACCESS = Q(is_active=True, learner__is_active=True)

#: A cohort registration grants access while it and its cohort are both active. The
#: member's own activity is tested where the member is looked up: see held_by_active_member.
COHORT_REGISTRATION_GRANTS_ACCESS = Q(is_active=True, cohort__is_active=True)


def held_by_active_member(member_filter: Q) -> Q:
    """Both member conditions in one Q, so they land on one membership join.

    Split across two filter() calls, a cohort holding both a removed and an
    active Learner for the same user would leak access through the removed one.
    """
    return member_filter & Q(cohort__cohortmembership__learner__is_active=True)


def access_granting_cohort_registrations() -> QuerySet[CohortCourseRegistration]:
    ...  # docstring unchanged
    return CohortCourseRegistration.objects.filter(COHORT_REGISTRATION_GRANTS_ACCESS)


def is_registered_for_course_expression(user: RequestUser) -> Q:
    ...  # docstring unchanged
    return Exists(
        LearnerCourseRegistration.objects.filter(
            LEARNER_REGISTRATION_GRANTS_ACCESS, course=OuterRef("pk"), learner__user=user
        )
    ) | Exists(
        access_granting_cohort_registrations().filter(
            held_by_active_member(Q(cohort__cohortmembership__learner__user=cast("User", user))),
            course=OuterRef("pk"),
        )
    )


def registered_course_pairs(site_id: int, user_ids: Iterable[int]) -> set[tuple[int, UUID]]:
    """Every (user_id, course_id) these users are registered for, by the dashboard's own test.

    _base_manager with an explicit site: this runs from tasks and receivers with
    no request, and must give the same answer with one.
    """
    ids = list(user_ids)
    individual = LearnerCourseRegistration._base_manager.filter(
        LEARNER_REGISTRATION_GRANTS_ACCESS, site_id=site_id, learner__user_id__in=ids
    ).values_list("learner__user_id", "course_id")
    through_cohort = CohortCourseRegistration._base_manager.filter(
        COHORT_REGISTRATION_GRANTS_ACCESS,
        held_by_active_member(Q(cohort__cohortmembership__learner__user_id__in=ids)),
        site_id=site_id,
    ).values_list("cohort__cohortmembership__learner__user_id", "course_id")
    return set(individual) | set(through_cohort)
```

`values_list` reuses the membership join the `filter()` made, so the user id it returns is the
member's. The test with a second, unrelated member in the same cohort pins that down.

### The snapshot mixin (`freedom_ls/learner_management/models.py`)

Lift `loaded_is_active` and `from_db` out of `LearnerCourseRegistration` into
`LoadedIsActiveMixin`, and put the mixin first in the bases of `LearnerCourseRegistration`,
`Learner` and `Cohort`. No field changes, so `makemigrations --check` stays clean.

```python
class LoadedIsActiveMixin:
    """Remember the is_active a row was loaded with.

    A save compares it with the new value to tell a flip from a no-op without a
    second query. None for a row built in memory or loaded with is_active deferred.
    """

    is_active: bool
    loaded_is_active: bool | None = None

    @classmethod
    def from_db(cls, db: str | None, field_names: Collection[str], values: Collection[object]) -> Self:
        instance = super().from_db(db, field_names, values)
        loaded = dict(zip(field_names, values, strict=False))
        if "is_active" in loaded:
            instance.loaded_is_active = bool(loaded["is_active"])
        return instance
```

The bare `is_active: bool` annotation declares the field every model using the mixin has, so a
receiver can read it without `getattr`. If the type checker rejects `super().from_db` on a plain
mixin, use the `TYPE_CHECKING` base pattern `freedom_ls/site_aware_models/forms.py` uses for
`_ModelFormBase`: the base is `models.Model` for the checker and `object` at runtime.

### Clearing (`freedom_ls/course_recommendations/queries.py`)

```python
def clear_registered_recommendations(site_id: int, user_ids: Iterable[int]) -> int:
    """Delete these users' recommendations for courses they are registered for, whoever made them.

    The rows are locked first, in pk order. A rule evaluation re-reads a
    recommendation under the same lock before attaching itself to it, so
    neither side commits a provenance row that points at a deleted recommendation.
    """
    ids = list(user_ids)
    registered = registered_course_pairs(site_id, ids)
    if not registered:
        return 0
    with transaction.atomic():
        locked = (
            RecommendedCourse._base_manager.select_for_update()
            .filter(site_id=site_id, user_id__in=ids, course_id__in={course for _, course in registered})
            .order_by("pk")
            .values_list("pk", "user_id", "course_id")
        )
        stale = [pk for pk, user_id, course_id in locked if (user_id, course_id) in registered]
        if stale:
            RecommendedCourse._base_manager.filter(pk__in=stale).delete()
        return len(stale)
```

```python
def _flipped_on(instance: LoadedIsActiveMixin, update_fields: frozenset[str] | None) -> bool:
    """A save that turned is_active on. A save that leaves it out of update_fields
    decides nothing, whatever a stale instance holds."""
    if update_fields is not None and "is_active" not in update_fields:
        return False
    return instance.loaded_is_active is False and instance.is_active
```

## Slice 3: a recommend rule puts a course in front of matching users, and registers nobody

### Evaluation (`freedom_ls/registration_rules/evaluation.py`)

One switch helper replaces the two copies of the per-organisation cache in `_hand_overs` and
`evaluate_learner`, and serves every new caller:

```python
def switched_on(rules: Iterable[RegistrationRule]) -> Iterator[RegistrationRule]:
    """The rules whose organisation's switch is on, read once per organisation.

    A learner who moved still holds the old organisation's rows, and that
    organisation has its own switch.
    """
    switches: dict[UUID, bool] = {}

        if switches[rule.organisation_id]:
```

Register paths see only register rules: `_hand_overs` adds
`outcome=RegistrationRule.Outcome.REGISTER` to the `others` filter, and `evaluate_rule` and
`evaluate_learner` dispatch on `rule.outcome` before calling a classify/apply pair, so
`classify_rule_changes` and `apply_rule_changes` are never handed a recommend rule.

```python
def _registered_users(state: RuleState, user_ids: Iterable[int]) -> set[int]:
    """A registered user counts as not matching: the dashboard already shows them the course."""

        user_id
        for user_id, course_id in registered_course_pairs(state.site_id, user_ids)

def classify_recommendation_changes(
    rule: RegistrationRule, state: RuleState, user_ids: Sequence[int] | None = None
) -> RecommendationChanges:
    """Compare the rule's match with the course's recommendations. Reads only."""
    matching = get_rule_kind(rule.kind).matching_learners(state)
    recommendations = RecommendedCourse._base_manager.filter(
        site_id=state.site_id, course_id=state.course_id
    )
    if user_ids is not None:
        matching = matching.filter(user_id__in=user_ids)
        recommendations = recommendations.filter(user_id__in=user_ids)
    matched = set(matching.values_list("user_id", flat=True))
    matched -= _registered_users(state, matched)
    # One entry per recommendation; a rule-made row appears once per rule, a
    # staff-made row once with None.
    rules_by_user: dict[int, set[UUID | None]] = defaultdict(set)
    for user_id, rule_id in recommendations.values_list("user_id", "rule_recommendations__rule_id"):
        rules_by_user[user_id].add(rule_id)
    recommend = sorted(matched - rules_by_user.keys())
    remove = sorted(
        user_id
        for user_id, rule_ids in rules_by_user.items()
        if rule_ids == {rule.pk} and user_id not in matched
    )
    return RecommendationChanges(recommend=tuple(recommend), remove=tuple(remove))
```

The write:

```python
def apply_recommendation_changes(
    rule: RegistrationRule, changes: RecommendationChanges
) -> EvaluationResult:
    """Write the changes, in user-pk order, with explicit site ids throughout."""
    recommended: list[int] = []
    for user_id in changes.recommend:
        try:
            # A savepoint per row: a recommendation made by someone else between
            # the read and this write hits the unique constraint, and only this
            # row rolls back. Theirs stands; a rule never takes over a row.
            # Explicit site_id: a task has no request for the model to read it from.
            with transaction.atomic():
                recommendation = RecommendedCourse._base_manager.create(
                    site_id=rule.site_id, user_id=user_id, course_id=rule.course_id
                )
                RuleRecommendation._base_manager.create(
                    site_id=rule.site_id, recommendation=recommendation, rule=rule
                )
        except IntegrityError:
            continue
        recommended.append(user_id)

    removed: list[int] = []
    # Re-read under a lock: a clearing receiver may have deleted the row, or
    # another rule may have attached itself to it, since the classifying read.
    for recommendation in (
        RecommendedCourse._base_manager.select_for_update()
        .filter(site_id=rule.site_id, course_id=rule.course_id, user_id__in=changes.remove)
        .order_by("pk")
        .prefetch_related("rule_recommendations")
    ):
        held_by = {held.rule_id for held in recommendation.rule_recommendations.all()}
        if held_by != {rule.pk}:
            continue  # another rule still holds this recommendation, so it stays
        recommendation.delete()
        removed.append(recommendation.user_id)
    return EvaluationResult(recommended=tuple(recommended), recommendations_removed=tuple(removed))
```

`evaluate_rule`: each outcome has a scope and a step; the batch loop and the

```python

    ...  # the existing in_scope union, moved here

    """Every user the rule matches plus every user it holds a RuleRecommendation for."""
    return sorted(
        set(get_rule_kind(rule.kind).matching_learners(state).values_list("user_id", flat=True))
        | set(

        )

    return apply_rule_changes(
        rule, classify_rule_changes(rule, RuleState.from_rule(rule), learner_ids=learner_ids)
    )


def _recommend_step(rule: RegistrationRule, user_ids: list[int]) -> EvaluationResult:
    return apply_recommendation_changes(
        rule, classify_recommendation_changes(rule, RuleState.from_rule(rule), user_ids=user_ids)
    )


def _run_in_batches[PK: (UUID, int)](
    rule_id: UUID,
    site_id: int,
    started_at: datetime,
    in_scope: Sequence[PK],
    step: Callable[[RegistrationRule, list[PK]], EvaluationResult],
) -> EvaluationResult:
    result = EvaluationResult()
    for batch in itertools.batched(in_scope, config.REGISTRATION_RULE_BATCH_SIZE, strict=False):
        # One transaction per batch, re-reading the rule under a row lock, so a
        # rule disabled or a switch turned off mid-run takes effect at the next
        # batch rather than after the whole organisation.
        with transaction.atomic():
            rule = (
                RegistrationRule._base_manager.select_for_update(no_key=True)
                .filter(pk=rule_id, site_id=site_id)
                .first()
            )
            if rule is None or not rules_enabled_for(rule.organisation_id):
                return result
            result = result.merged_with(step(rule, list(batch)))
    # update(), not save(): a finished run is not an edit of the rule.
    RegistrationRule._base_manager.filter(pk=rule_id, site_id=site_id).update(
        last_evaluated_at=started_at
    )
    return result


def evaluate_rule(rule_id: UUID, site_id: int) -> EvaluationResult:
    """Reconcile one rule against everyone it matches or holds."""
    ...  # the existing guard, started_at and state
    if rule.outcome == RegistrationRule.Outcome.RECOMMEND:
        return _run_in_batches(rule_id, site_id, started_at, _recommend_scope(rule, state), _recommend_step)
    return _run_in_batches(rule_id, site_id, started_at, _register_scope(rule, state), _register_step)
```

## Slice 4: rules share a recommendation without taking one over

**Behaviour:** a user matched by rules in two organisations has one recommendation with two
`RuleRecommendation`s. One rule unmatching drops only its own row; the last one out deletes the
recommendation. A staff-made recommendation never gains a `RuleRecommendation`. A user whose
recommendations a rule holds is re-evaluated when their `Learner` changes, even in another
organisation.

### `RecommendationChanges` grows two groups (`evaluation.py`)

- staff-made (`{None}`): left alone.

- `drop_rule_recommendation`: `rule.pk in rule_ids and len(rule_ids) > 1` and the user not

## Slice 5: a user who stops being registered is recommended again

**Behaviour:** when a registration, membership, cohort or `Learner` that gave a user a course is
deactivated or deleted, every enabled recommend rule on that course whose switch is on
re-evaluates them. A register rule retracting a learner also triggers it.

### Receivers (`freedom_ls/registration_rules/signals.py`)

A shared resolver and two enqueue shapes:

```python
def _recommend_rules_on(site_id: int, course_ids: Iterable[UUID]) -> list[RegistrationRule]:
    """Enabled recommend rules on these courses whose organisation's switch is on."""
    return list(
        switched_on(
            RegistrationRule._base_manager.filter(
                site_id=site_id,
                course_id__in=course_ids,
                is_enabled=True,
                outcome=RegistrationRule.Outcome.RECOMMEND,
            ).order_by("pk")
        )
    )


def _re_recommend_user(site_id: int, user_id: int, course_ids: Iterable[UUID]) -> None:
    """One evaluate_learner per active Learner of the user in a rule's organisation."""
    rules = _recommend_rules_on(site_id, course_ids)
    for learner_id in Learner._base_manager.filter(
        site_id=site_id, user_id=user_id, is_active=True,
        organisation_id__in={rule.organisation_id for rule in rules},
    ).values_list("pk", flat=True):
        _enqueue_learner(learner_id, site_id)


def _re_recommend_cohort(site_id: int, course_ids: Iterable[UUID]) -> None:
    """A cohort-wide change: one run per rule, not one task per member."""
    for rule in _recommend_rules_on(site_id, course_ids):
        _enqueue_rule(rule.pk, site_id)
```

One receiver per model, stacked `@receiver(post_save, ...)` and `@receiver(pre_delete, ...)` as
`protect_entry_a_rule_names` stacks its three, so the resolve step is written once. The
`post_save` branch returns early unless the save withdrew access; the `pre_delete` branch always
resolves, because the cascade may have removed the rows by `post_delete`. Every receiver skips
`raw=True` and goes through `_enqueue_once`, which already runs after commit:

| Receiver on | Withdrawal on `post_save` | Resolves | Enqueues |
| --- | --- | --- | --- |
| `LearnerCourseRegistration` | `not instance.is_active` | the learner's user, that course | `_re_recommend_user` |
| `CohortMembership` | never (a membership is only created or deleted) | the member's user, the cohort's registered courses (`CohortCourseRegistration._base_manager.filter(cohort_id=...)`) | `_re_recommend_user` |
| `Learner` | `loaded_is_active is True and not is_active`, skipping an `update_fields` without `is_active` | the user and the courses registered through this `Learner` (individual and cohort), resolved before the cascade; on delete also every rule with a `RuleRecommendation` on the user's recommendations | `_re_recommend_user`; on delete also `_enqueue_rule` per rule |
| `CohortCourseRegistration` | `not instance.is_active` | that course | `_re_recommend_cohort` |
| `Cohort` | `loaded_is_active is True and not is_active`, same `update_fields` skip; no `pre_delete` branch, the memberships' and registrations' own receivers fire on the cascade | the cohort's registered courses | `_re_recommend_cohort` |

The existing `Learner` `post_save` trigger (`evaluate_learner_on_save`) already evaluates the
deactivated `Learner` itself, which removes the rule's recommendation when no other `Learner` of
the user matches. `detach_rule_on_foreign_is_active_change` is a `pre_save` and stays as it is;
the new registration receiver is a `post_save`, so a register rule's retraction (`is_active`
flipped to False by `_set_is_active`) fires it too, as intended.

## Slice 6: deleting a recommend rule

**Behaviour:** a disabled recommend rule cannot be deleted while its removal run is pending; a

### `_delete_refusal` (`admin.py`)

```python
        return f"Rule still removing its recommendations: {rule.name}"
    return None
```

```python
def solely_held_by(rule: RegistrationRule) -> QuerySet[RecommendedCourse]:
```

## Slice 8: the recommendation admin knows who made a row

**Behaviour:** `RecommendedCourseAdmin` shows which rules made a row, filters by it, and warns
before deleting a rule-made one. Rule names and links show only to a user with
`view_registrationrule`; anyone else sees a rule-made row as "Registration rule" with no name.

### Wiring (bottom of `registration_rules/admin.py`)

```python
_VIEW_RULE = "freedom_ls_registration_rules.view_registrationrule"


def _rules_of(obj: RecommendedCourse) -> list[RegistrationRule]:
    return sorted(
        (rule_recommendation.rule for rule_recommendation in obj.rule_recommendations.all()),
        key=lambda rule: rule.name,
    )


def registration_rules_display(can_view_rules: bool) -> Callable[[RecommendedCourse], str | SafeString]:
    """The read-only "Registration rules" field, bound to one request's permission.

    Named and linked only for someone allowed to view rules: a rule may belong
    to an organisation this person may not see. Bound per request because the
    admin instance is shared by every request, so nothing per user may live on it.
    """

    @admin.display(description="Registration rules")
    def registration_rules(obj: RecommendedCourse) -> str | SafeString:
        rules = _rules_of(obj)
        if not rules:
            return ""
        if not can_view_rules:
            return "Registration rule"
        return format_html_join(
            ", ",
            '<a href="{}">{}</a>',
            (
                (reverse("admin:freedom_ls_registration_rules_registrationrule_change", args=[rule.pk]), rule.name)
                for rule in rules
            ),
        )

    return registration_rules


def made_by_display(can_view_rules: bool) -> Callable[[RecommendedCourse], str]:
    """The "Made by" column, bound to one request's permission for the same reason."""

    @admin.display(description="Made by")
    def made_by(obj: RecommendedCourse) -> str:
        rules = _rules_of(obj)
        if not rules:
            return "Staff"
        if not can_view_rules:
            return "Registration rule"
        return ", ".join(rule.name for rule in rules)

    return made_by


class MadeByFilter(admin.SimpleListFilter):
    title = "Made by"
    parameter_name = "made_by"

    def lookups(self, request, model_admin):
        return [("rule", "Registration rule"), ("staff", "Staff")]

    def queryset(self, request, queryset):
        rule_made = Exists(RuleRecommendation._base_manager.filter(recommendation=OuterRef("pk")))
        if self.value() == "rule":
            return queryset.filter(rule_made)
        if self.value() == "staff":
            return queryset.exclude(rule_made)
        return queryset
```

A display method receives no request, and the admin instance is shared across requests, so the
permission is bound per request where Django hands the request over: a wrapped
`get_list_display(request)` returns `[*stock, made_by_display(request.user.has_perm(_VIEW_RULE))]`
and a wrapped `get_readonly_fields(request, obj)` returns
`[*stock, registration_rules_display(...)]`; both accept callables. A wrapped `get_queryset`
adds `prefetch_related("rule_recommendations__rule")` so the column costs no query per row.

`get_deleted_objects` is wrapped: on the confirmation request (`GET`, or a `POST` without
`post` in it, which is the bulk action's first step) and when any selected row has a
`RuleRecommendation`, `messages.warning(request, "A recommendation a registration rule made comes back on the rule's next evaluation while the user still matches. Disable or change the rule instead.")`.
The wiring follows the `LearnerCourseRegistrationAdmin` block: `list_filter = [*..., MadeByFilter]`
on the class, and `setattr` for `get_list_display`, `get_readonly_fields`, `get_queryset` and
`get_deleted_objects`, each wrapping the stock method it replaces.

### Tests (`freedom_ls/registration_rules/tests/test_admin.py`)
