# Implementation plan: referral codes and the redirects behind them

## Goal

Extend `freedom_ls/referral_tracking` as `1. spec.md` describes: a `ReferralCode` a builder creates
in the admin, a `ReferralCodeHit` log, the `/go/{code}` and `/d/{CODE}` redirect view, `ref` as a
tracked parameter that lands in `referral_code` on `SignupAttribution` and `FirstTouchCount`,
capture suppression on the redirect routes, a writable `ReferralCode` admin with a copy button and a
deactivate action, a read-only `ReferralCodeHit` admin, one setting with a system check, and the
`prune_referral_code_hits` command.

Every design decision is settled in the spec and is not reopened here. The spec's vocabulary is
this plan's vocabulary: referral code, hit, door, machine fetch, destination, inactive destination,
site path, first touch, attribution key, mint, tally, capture.

The spec's "Documentation and downstream" section (upgrade notes, product docs, the glossary
section, the app map) is written after implementation by its own steps and is out of this plan's
scope on purpose.

```python
def full_clean(self, *args: object, **kwargs: object) -> None:
    self._set_site_from_request()
    if not self.code and self.site_id:
        self.code = generate_code(self.site)
    super().full_clean(*args, **kwargs)

def clean(self) -> None:
    super().clean()
    if self.site_id:
        try:
            validate_code_text(self.code, self.site)
        except ValidationError as exc:
            raise ValidationError({"code": exc.messages}) from exc
```

  `full_clean` is the hook because `clean_fields()` runs before `clean()` and would reject the blank
  `code` first. The base `full_clean` sets `site` from the request and is called again harmlessly.
  `ModelForm._post_clean()` constructs the instance, calls `full_clean()`, then
  `validate_constraints()`, so the constraint sees the generated value and a collision is a form
  error, never an `IntegrityError`. `codes.py` imports `ReferralCode`, so `models.py` imports
  `generate_code` and `validate_code_text` inside the two methods.
- `__str__` on `ReferralCode` returns `self.code`; on `ReferralCodeHit` returns
  `f"{self.referral_code_id} via {self.door} at {self.hit_at:%Y-%m-%d}"`.

### Task 2: code text and paths (`codes.py`, new)

`freedom_ls/referral_tracking/codes.py` holds every constant and function the spec names under
*Code text* and *Paths*, plus the lookup.

```python
CODE_PATTERN = re.compile(r"[A-Za-z0-9-]{1,64}")
GENERATED_CODE_ALPHABET = "0123456789ABCDEFGHJKMNPQRSTVWXYZ"  # pragma: allowlist secret
GENERATED_CODE_LENGTH = 8
GENERATED_CODE_ATTEMPTS = 20
RESERVED_REFERRAL_CODES = frozenset({"admin", "api", "account", "accounts", "login", "logout",
    "signup", "register", "support", "help", "contact", "security", "billing", "official",
    "staff", "static", "media", "www", "mail"})


def validate_code_text(code: str, site: Site) -> None:
    if not CODE_PATTERN.fullmatch(code):
        raise ValidationError("Use 1 to 64 letters, digits or hyphens.")
    lowered = code.lower()
    if lowered in RESERVED_REFERRAL_CODES:
        raise ValidationError(f"{code!r} is reserved. Choose a different code.")
    brand = slugify(site.name) if site.name else ""
    if brand and lowered in {brand, brand.replace("-", "")}:
        raise ValidationError("The site's own name is reserved. Choose a different code.")


def lookup_referral_code(site: Site, code: str) -> ReferralCode | None:
    # _base_manager, not `objects`: the site-aware manager also ANDs in the
    # ambient request's site, and generate_code may be probing another.
    return (
        ReferralCode._base_manager.filter(site=site)
        .alias(code_lower=Lower("code"))
        .filter(code_lower=code.lower())
        .first()
    )


def generate_code(site: Site) -> str:
    for _ in range(GENERATED_CODE_ATTEMPTS):
        candidate = "".join(secrets.choice(GENERATED_CODE_ALPHABET) for _ in range(GENERATED_CODE_LENGTH))
        try:
            validate_code_text(candidate, site)
        except ValidationError:
            continue
        if lookup_referral_code(site, candidate) is None:
            return candidate
    raise ValidationError("Could not find an unused code. Save again.")


def validate_site_path(value: str) -> None:
    if not value.startswith("/") or value[1:2] in ("/", "\\"):
        raise ValidationError("Enter a path on this site, starting with a single '/'.")
    if any(ch.isspace() or ord(ch) < 32 or ord(ch) == 127 for ch in value):
        raise ValidationError("A path cannot contain whitespace or control characters. Remove them.")
    path = urlsplit(value).path
    try:
        match = resolve(path)
    except Resolver404:
        return
    if match.func is follow_referral_code:
        raise ValidationError("A destination cannot be a referral code route. Point it at a page.")


def inactive_destination_for(referral_code: ReferralCode) -> str:
    return referral_code.inactive_destination or config.REFERRAL_TRACKING_INACTIVE_DESTINATION


def build_redirect_url(referral_code: ReferralCode, query_string: str, *, base: str | None = None) -> str:
    if base is None:
        base = referral_code.destination if referral_code.is_active else inactive_destination_for(referral_code)
    parts = urlsplit(base)
    visitor = [(k, v) for k, v in parse_qsl(query_string, keep_blank_values=True) if k != "ref"]
    visitor_keys = {k for k, _ in visitor}
    kept = [(k, v) for k, v in parse_qsl(parts.query, keep_blank_values=True)
            if k != "ref" and k not in visitor_keys]
    query = urlencode([*kept, *visitor, ("ref", referral_code.code)])
    return urlunsplit((parts.scheme, parts.netloc, parts.path, query, parts.fragment))


def absolute_code_url(referral_code: ReferralCode, door: Door) -> str:
    protocol = getattr(settings, "ACCOUNT_DEFAULT_HTTP_PROTOCOL", "https")
    url_name = "referral_tracking:follow_go" if door == Door.GO else "referral_tracking:follow_d"
    path = reverse(url_name, kwargs={"code": referral_code.code})
    url = f"{protocol}://{referral_code.site.domain}{path}"
    return url.upper() if door == Door.D else url
```

`validate_site_path` imports `follow_referral_code` from `views.py` inside the function body, since
`views.py` imports `codes.py`. `validate_site_path` is a plain validator; the model fields
`destination` and `inactive_destination` list it in `validators=[validate_site_path]` so the admin
form reports a bad path on the field itself, and the check in Task 7 calls it on the setting.

Tests (`tests/test_codes.py`): the spec's *Code text*, *Paths* and *`build_redirect_url`* lists,
one behaviour per test. Reserved words and the brand cases are `parametrize`d. Generation retries
past a collision (build the colliding row with `ReferralCodeFactory`, patch `secrets.choice` at the
module boundary to return a fixed sequence) and past a reserved candidate. `absolute_code_url` uses
the code's site domain and uppercases the whole `Door.D` string.

### Task 3: hits (`hits.py`, new)

```python
NAMED_FETCHERS = ("slackbot", "facebookexternalhit", "twitterbot", "whatsapp", "telegrambot",
    "discordbot", "linkedinbot", "googlebot", "bingbot", "applebot", "skypeuripreview")
GENERIC_TOKENS = ("crawler", "spider", "preview", "headless")
BOT_WORD = re.compile(r"\bbot\b")
PREFETCH_HEADERS = ("Sec-Purpose", "Purpose", "X-Moz")


def is_machine_fetch(request: HttpRequest) -> bool:
    user_agent = request.headers.get("User-Agent", "").lower()
    if not user_agent:
        return True
    if any(name in user_agent for name in NAMED_FETCHERS):
        return True
    if any(token in user_agent for token in GENERIC_TOKENS) or BOT_WORD.search(user_agent):
        return True
    return any("prefetch" in request.headers.get(header, "").lower() for header in PREFETCH_HEADERS)


def record_hit(referral_code: ReferralCode, door: Door, request: HttpRequest) -> None:
    machine_fetch = is_machine_fetch(request)
    try:
        with transaction.atomic():
            ReferralCodeHit._base_manager.create(
                site=referral_code.site, referral_code=referral_code,
                door=door, is_machine_fetch=machine_fetch,
            )
            if not machine_fetch:
                ReferralCode._base_manager.filter(pk=referral_code.pk).update(
                    hit_count=F("hit_count") + 1, last_hit_at=timezone.now()
                )
    except DatabaseError as exc:
        sentry_sdk.capture_exception(exc)
```

The site is passed explicitly, from the code's own row, so a hit can never land on another tenant
even when the thread-local request resolves differently.

Tests (`tests/test_hits.py`): each named fetcher, an empty UA and each prefetch header are
flagged (`parametrize` over `RequestFactory` requests); a fixed list of real browser UAs including a
CUBOT Android string is not; a flagged hit writes a row and leaves `hit_count` and `last_hit_at`
alone; an unflagged hit moves both; a `DatabaseError` raised by patching
`ReferralCodeHit._base_manager.create` is reported to `sentry_sdk.capture_exception` and not raised.
Patching a manager's `create` is the one place this plan mocks an ORM call. It copies the
`failing_attribution_write` fixture in `tests/test_signup.py:164`, the app's existing way of
driving the savepoint-and-report path, so the two best-effort writes are tested alike.

```python
@require_safe
def follow_referral_code(request: HttpRequest, code: str, door: Door) -> HttpResponse:
    suppress_capture(request)
    site = get_cached_site(request)
    if not isinstance(site, Site):
        raise Http404
    referral_code = lookup_referral_code(site, code)
    if referral_code is None:
        raise Http404
    if request.method == "GET":
        record_hit(referral_code, door, request)
    query_string = request.META.get("QUERY_STRING", "")
    target = build_redirect_url(referral_code, query_string)
    if not url_has_allowed_host_and_scheme(target, allowed_hosts={request.get_host()}):
        target = build_redirect_url(referral_code, query_string, base="/")
    response = HttpResponseRedirect(target)
    response["Cache-Control"] = "private, no-store"
    response["X-Robots-Tag"] = "noindex"
    return response
```

`require_safe` answers `POST` with 405 before the view body runs; the middleware ignores non-GET
requests anyway, so no capture is possible on that path.

    self, request: HttpRequest, queryset: QuerySet[ReferralCode]
) -> None:
    changed = queryset.filter(is_active=True).update(is_active=False)
    self.message_user(request, f"Deactivated {changed} referral code(s).")

@admin.display(description="Go URL")
def go_url(self, obj: ReferralCode) -> str:
    return self._copyable_url(obj, Door.GO)

@admin.display(description="D URL")
def d_url(self, obj: ReferralCode) -> str:
    return self._copyable_url(obj, Door.D)

@admin.display(description="Destination preview")
def destination_preview(self, obj: ReferralCode) -> str:
    if obj._state.adding:
        return UNSAVED_MESSAGE
    return build_redirect_url(obj, "", base=obj.destination)

@admin.display(description="Inactive destination preview")
def inactive_destination_preview(self, obj: ReferralCode) -> str:
    if obj._state.adding:
        return UNSAVED_MESSAGE
    preview = build_redirect_url(obj, "", base=inactive_destination_for(obj))
    return preview if obj.inactive_destination else f"{preview} (the default)"

def _copyable_url(self, obj: ReferralCode, door: Door) -> str:
    if obj._state.adding:
        return UNSAVED_MESSAGE
    url = absolute_code_url(obj, door)
    element_id = f"referral-code-{door}-url"
    return format_html(
        '<span id="{}">{}</span> '
        '<button type="button" class="referral-code-copy" data-copy-target="{}" '
        'aria-label="Copy {} URL">Copy</button>',
        element_id, url, element_id, door.label,
    )
```

  `UNSAVED_MESSAGE = "Shown once the code is saved."` is a module constant. `has_delete_permission`
  returning `False` is what removes Django's "Delete selected" action. `format_html` escapes the
  URL and the ids; nothing else needs `mark_safe`. Unfold's `ModelAdmin` renders a `SafeString`
  returned by a read-only method as HTML, as `OrganisationAdmin.contributed_summaries` relies on.

  admin class. Do neither.

function init() {
  var buttons = document.querySelectorAll("button[data-copy-target]");
  if (!buttons.length) return;
  var live = document.createElement("div");
  live.setAttribute("aria-live", "polite");
  live.className = "sr-only";
  document.body.appendChild(live);
  buttons.forEach(function (button) {
    button.addEventListener("click", function () {
      var target = document.getElementById(button.dataset.copyTarget);
      if (!target || !navigator.clipboard) return;
      navigator.clipboard.writeText(target.textContent).then(function () {
        live.textContent = "Copied";
      });
    });
  });
}
```

```python
PRUNE_BATCH_SIZE = 1000

@click.command()
@click.option("--older-than-days", type=click.IntRange(min=1), required=True)
def command(older_than_days: int) -> None:
    """Delete referral code hits older than the cutoff, on every site."""
    cutoff = timezone.now() - timedelta(days=older_than_days)
    deleted = 0
    while True:
        batch = list(
            ReferralCodeHit._base_manager.filter(hit_at__lt=cutoff)
            .order_by("pk").values_list("pk", flat=True)[:PRUNE_BATCH_SIZE]
        )
        if not batch:
            break
        deleted_count, _ = ReferralCodeHit._base_manager.filter(pk__in=batch).delete()
        deleted += deleted_count
    click.echo(f"Deleted {deleted} referral code hit(s) older than {older_than_days} day(s).")
```

`_base_manager` says in code what the spec says in words: every site, no ambient filter. The
command never touches `hit_count` or `last_hit_at`.

cutoff on two sites are deleted (build them with `ReferralCodeHitFactory` and move `hit_at` back

unchanged, and `--older-than-days 0` is refused.

## Naming

- `codes.py` holds what concerns the code text and its paths; `hits.py` holds what concerns one
  access. Neither module imports the other.
- `lookup_referral_code` is the spec's "look up"; `inactive_destination_for` is step 1's inactive
  branch; `base` is the spec's "base path". All three are named in the spec.
- The query parameter is `ref`; the cookie key is `rc`; the column, the factory attribute, the
  export column and the admin column are all `referral_code`. Only the parameter name differs, and
  `first_touch_from_request` is the one line that bridges it.
- `Door.GO` and `Door.D` are passed as URL kwargs and stored as `ReferralCodeHit.door`. The word
  "door" appears in no URL and no user-facing string; the labels are `/go/` and `/d/`.
