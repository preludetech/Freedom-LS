## Slice 1: cohort registration changes reach integrators

**Behaviour:** registering a cohort for a course records a `cohort.course_registered` event from a
command with no request; unregistering records `cohort.course_unregistered`; repeating either
records nothing.

1. `freedom_ls/webhooks/events.py`:

   ```python
   def fire_webhook_event(
       event_type: str, payload: dict[str, object], site: Site | None = None
   ) -> None:
       """The site comes from the caller when there is no request to read it
       from (an action run from a command or a task); otherwise from the
       request, and with neither the event is dropped.
       """
       validate_event_type(event_type)
       if site is None:
           request = getattr(_thread_locals, "request", None)
           if request is None:
               return
           resolved = get_cached_site(request)
           if not isinstance(resolved, Site):
               return
           site = resolved
       event = WebhookEvent.objects.create(event_type=event_type, payload=payload, site_id=site.pk)
       default_task_backend.enqueue(_dispatch_event_task, args=[str(event.pk), site.pk], kwargs={})
   ```

2. `freedom_ls/base/webhook_event_types.py`: append `("cohort.course_registered", "Cohort
   registered for course")` and `("cohort.course_unregistered", "Cohort unregistered from course")`
   to `FLS_WEBHOOK_EVENT_TYPES`, with samples in `WEBHOOK_EVENT_TYPE_SAMPLES` carrying
   `organisation_id`, `cohort_id`, `course_id`, `course_title`.
3. `freedom_ls/learner_management/utils.py`:

   ```python
   def fire_webhook_event_on_commit(
       event_type: str, payload: dict[str, object], site: Site
   ) -> None:
       """Every learner-management event fires once the transaction it belongs
       to has committed, so an integrator never hears about a rolled-back change."""
       from freedom_ls.webhooks.events import fire_webhook_event

       transaction.on_commit(lambda: fire_webhook_event(event_type, payload, site=site))


   def announce_cohort_registration_change(registration: CohortCourseRegistration) -> None:
       """Tell integrators a cohort registration changed. Called only by a
       change, never by a no-op save."""
       cohort = registration.cohort
       event_type = (
           "cohort.course_registered" if registration.is_active else "cohort.course_unregistered"
       )
       fire_webhook_event_on_commit(
           event_type,
           {
               "organisation_id": str(cohort.organisation_id),
               "cohort_id": str(cohort.pk),
               "course_id": str(registration.course_id),
               "course_title": registration.course.title,
           },
           cohort.site,
       )
   ```

   `learner_management → webhooks` edge off module load, the way `learner_progress/signals.py` and
   `accounts` already import it; `transaction.on_commit` outside `atomic` runs at once.

1. New `freedom_ls/accounts/account_setup.py`, the first two functions:

   ```python
   def pending_user_q(prefix: str = "") -> Q:
       """Q matching users who have never logged in and cannot: staff-created
       accounts whose setup link is still the only way in."""
       return Q(**{
           f"{prefix}last_login__isnull": True,
           f"{prefix}password__startswith": UNUSABLE_PASSWORD_PREFIX,
       })


   def is_pending(user: User) -> bool:
       return user.last_login is None and not user.has_usable_password()
   ```

   `UNUSABLE_PASSWORD_PREFIX` from `django.contrib.auth.hashers`.
2. `freedom_ls/learner_management/queries.py`:

   ```python
   class LearnerStatus(TextChoices):
       ACTIVE = "active", "Active"
       PENDING = "pending", "Pending"
       REMOVED = "removed", "Removed"


   def learner_status(learner: Learner) -> LearnerStatus:
       if not learner.is_active:
           return LearnerStatus.REMOVED
       if is_pending(learner.user):
           return LearnerStatus.PENDING
       return LearnerStatus.ACTIVE


   def learner_status_q(status: LearnerStatus) -> Q:
       """The queryset twin of learner_status, on a Learner queryset."""
       pending = pending_user_q("user__")
       if status == LearnerStatus.REMOVED:
           return Q(is_active=False)
       if status == LearnerStatus.PENDING:
           return Q(is_active=True) & pending
       return Q(is_active=True) & ~pending
   ```

## Slice 8: deactivate and reactivate a learner

**Behaviour:** the header offers Deactivate for an active or pending learner and Reactivate for a

   ```python
   """The single seam for each learner administration action: one transaction,
   no request, webhook events on commit only when something changed."""


   def _learner_payload(learner: Learner) -> dict[str, object]:
       user = learner.user
       return {
           "user_id": user.pk, "user_email": user.email,
           "organisation_id": str(learner.organisation_id), "learner_id": str(learner.pk),
       }


   def _fire(event_type: str, learner: Learner, extra: dict[str, object] | None = None) -> None:
       fire_webhook_event_on_commit(
           event_type, {**_learner_payload(learner), **(extra or {})}, learner.site
       )


   def _set_learner_active(learner: Learner, is_active: bool, event_type: str) -> bool:
       with transaction.atomic():
           locked = Learner._base_manager.select_for_update().get(pk=learner.pk)
           if locked.is_active == is_active:
               return False
           locked.is_active = is_active
           locked.save(update_fields=["is_active"])
           learner.is_active = is_active
           _fire(event_type, locked)
       return True


   def deactivate_learner(learner: Learner) -> bool:
       return _set_learner_active(learner, False, "learner.deactivated")


   def reactivate_learner(learner: Learner) -> bool:
       return _set_learner_active(learner, True, "learner.reactivated")
   ```

   ```python
   def match_or_create_user(
       site: Site, email: str, first_name: str = "", last_name: str = ""

           "expires_in_days": max(1, timeout // 86400),
   ```

**Behaviour:** `move_between_cohorts` moves the membership in one transaction, carries the

   as sent synchronously inside the move's transaction with `learner`, `source`, `destination`.
2. `learner_actions.py`:

   ```python
   def move_between_cohorts(learner: Learner, source: Cohort, destination: Cohort) -> str:
       if source.pk == destination.pk:
           raise ValueError("The source and destination cohorts are the same.")
       _check_cohort(learner, source)
       _check_cohort(learner, destination)
       move_id = str(uuid4())
       with transaction.atomic():
           Learner._base_manager.select_for_update().get(pk=learner.pk)
           deleted, _ = CohortMembership._base_manager.filter(learner=learner, cohort=source).delete()
           if not deleted:
               raise ValueError("The learner is not a member of the source cohort.")
           if CohortMembership._base_manager.filter(learner=learner, cohort=destination).exists():
               raise ValueError("The learner is already a member of the destination cohort.")
           CohortMembership._base_manager.create(site=learner.site, learner=learner, cohort=destination)
           cohort_membership_moved.send(
               sender=Learner, learner=learner, source=source, destination=destination
           )
           _fire("cohort.member_removed", learner, _cohort_payload(source, move_id))
           _fire("cohort.member_added", learner, _cohort_payload(destination, move_id))
       return move_id
   ```


4. `learner_progress/signals.py`:

   ```python
   @receiver(cohort_membership_moved, dispatch_uid="learner_progress.cohort_membership_moved")
   def carry_course_progress_on_move(
       sender: type[Learner], learner: Learner, source: Cohort, destination: Cohort, **kwargs: object
   ) -> None:
       """Carry the learner's work across a move. With no record under the
       destination registration the source record moves there; where both
       exist, the further one ends up under the destination and the other
       under the source, unused; a tie leaves both. Runs inside the move's
       transaction so the membership's own on-commit mint finds the moved
       record."""
       for carry in plan_progress_carry(learner, source, destination):
           record, target, existing = carry.record, carry.destination_registration, carry.existing
           if existing is None:
               record.cohort_registration = target
               record.save(update_fields=["cohort_registration"])
           elif record.progress_percentage > existing.progress_percentage:
               CourseProgress._base_manager.filter(pk__in=[record.pk, existing.pk]).update(
                   cohort_registration=Case(
                       When(pk=record.pk, then=Value(target.pk)),
                       When(pk=existing.pk, then=Value(record.cohort_registration_id)),
                   )
               )
   ```
