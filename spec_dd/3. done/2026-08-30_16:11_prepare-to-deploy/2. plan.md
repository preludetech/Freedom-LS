```python
def get_client_ip(request: HttpRequest) -> str:
    """Return the client IP address.

    When `config.TRUSTED_PROXY_IP_HEADER` names a header (e.g. "X-Real-IP"), returns
    that header's value verbatim. The header must be one the edge *sets* rather than
    appends, so it carries exactly one address; the value is never split.
    Falls back to REMOTE_ADDR when no header is configured or the named header is
    absent, which is what keeps this correct with no proxy in front.

    This is the only sanctioned way to derive a client IP for LegalConsent records,
    django-axes lockouts and similar evidence trails.
    """
    header_name: str | None = config.TRUSTED_PROXY_IP_HEADER

    if header_name:
        value = request.headers.get(header_name, "")
        if value:
            return str(value)

    return str(request.META.get("REMOTE_ADDR", "") or "")
```

The docstring must not mention `X-Forwarded-For`. That is the header this function must never be
pointed at.

### Shape

djclick, thin `command()` over module-level helpers, exactly like `sync_role_permissions`:

```python
"""First-run bootstrap for a production deployment: the Site, the administrative
User and its verified allauth EmailAddress."""

import djclick as click
from allauth.account.models import EmailAddress

from django.conf import settings
from django.contrib.sites.models import Site
from django.utils.crypto import get_random_string

from freedom_ls.accounts.models import User

# 22 characters over get_random_string's 62-character alphabet is ~131 bits.
GENERATED_PASSWORD_LENGTH = 22


@click.command()
@click.argument("admin_email")
@click.option("--domain", default=None, help="...")
@click.option("--site-name", default=None, help="...")
def command(admin_email: str, domain: str | None, site_name: str | None) -> None:
    """Create the Site, the administrative User and its verified email address."""
    resolved_domain = _resolve_domain(domain)
    site = _get_or_create_site(resolved_domain, site_name or resolved_domain)
    password = _get_or_create_admin_user(admin_email, site)
    if password is not None:
        click.echo(f"Created administrator {admin_email} with password: {password}")
        click.echo("This password is shown once and is stored nowhere. Record it now.")
    else:
        click.echo(f"Administrator {admin_email} already exists; password unchanged.")


def _resolve_domain(domain: str | None) -> str:
    if domain:
        return domain
    host_domain = getattr(settings, "HOST_DOMAIN", "")
    if not host_domain:
        raise click.ClickException(
            "No --domain given and HOST_DOMAIN is not set in this settings module."
        )
    return str(host_domain)


def _get_or_create_site(domain: str, name: str) -> Site:
    # Domain is unique; name is not. A lookup keyed on name could match a row
    # whose domain differs from the one just resolved, and writing only the name
    # field would leave that wrong domain in place.
    site, _created = Site.objects.get_or_create(domain=domain, defaults={"name": name})
    return site


def _get_or_create_admin_user(email: str, site: Site) -> str | None:
    """Return the generated password when the User was created, else None.

    set_password is reachable only on the create branch, so a second run never
    resets an existing administrator's credential.

    _base_manager, not the site-aware `objects`: email is globally unique, and a
    lookup that narrowed to an ambient site would miss an existing administrator
    and try to create a second row with the same address.
    """
    existing = User._base_manager.filter(email=email).first()
    if existing is not None:
        _ensure_verified_email(existing)
        return None

    # UserManager.create_user/create_superuser take no site argument, so the row
    # is constructed directly.
    password = get_random_string(GENERATED_PASSWORD_LENGTH)
    user = User(email=email, site=site, is_staff=True, is_superuser=True, is_active=True)
    user.set_password(password)
    user.save()
    _ensure_verified_email(user)
    return password


def _ensure_verified_email(user: User) -> None:
    EmailAddress.objects.get_or_create(
        user=user, email=user.email, defaults={"verified": True, "primary": True}
    )
```

Constraints that must hold in the finished code:

- The password reaches `click.echo` and nothing else — no `logging`, no file, no column.
- Nothing creates an `Organisation`. The `post_save` receiver on `Site` does that.
- Every write is a lookup-or-create.

### `freedom_ls/deployment/worker.py`

```python
"""Heartbeat and watchdog for the database task worker.

HeartbeatWorker.run() is a copy of django_tasks_db's Worker.run() loop with one
line added. django-tasks-db is pinned exactly (==0.12.0) and a test asserts that
pin, so a version bump fails loudly and this copy gets re-checked.
"""

WATCHDOG_POLL_SECONDS = 30
SENTRY_FLUSH_TIMEOUT_SECONDS = 5


def touch_heartbeat(path: str | Path) -> None:
    """Update the file's mtime, creating it and its parents when absent.

    Only the mtime matters; the contents are never read.
    """
    heartbeat = Path(path)
    heartbeat.parent.mkdir(parents=True, exist_ok=True)
    heartbeat.touch()


def heartbeat_age_seconds(path: str | Path) -> float | None:
    """Seconds since the heartbeat's mtime, or None when the file is absent."""


def check_worker_heartbeat(path: str | Path, max_age_seconds: int) -> None:
    """One watchdog pass. Reports and kills the process when the heartbeat is stale.

    An absent file counts as stale: fls_run_worker touches the path before the
    watchdog starts, and the external probe's own -mmin window finds nothing either.
    """
    age = heartbeat_age_seconds(path)
    if age is not None and age <= max_age_seconds:
        return
    sentry_sdk.capture_message(
        f"fls_run_worker heartbeat is stale (>{max_age_seconds}s); exiting."
    )
    # Not optional. os._exit skips atexit, and the SDK's automatic drain is an
    # atexit integration, so without this the report is lost in exactly the case
    # that matters.
    sentry_sdk.flush(timeout=SENTRY_FLUSH_TIMEOUT_SECONDS)
    # Not sys.exit(): that raises SystemExit in this thread only and leaves the
    # wedged main thread running.
    os._exit(1)


def start_watchdog(
    path: str | Path,
    max_age_seconds: int,
    poll_seconds: int = WATCHDOG_POLL_SECONDS,
) -> threading.Thread:
    """Start the daemon watchdog thread and return it.

    daemon=True so a clean shutdown is never held open by this thread.
    """


class HeartbeatWorker(Worker):
    def run(self) -> None:
        """Copy of django_tasks_db 0.12.0 Worker.run() with a heartbeat touch."""
```

`HeartbeatWorker.run()` is a faithful copy of the upstream body — the `startup_delay` sleep, the
`while self.running:` loop, the queue filter, the `exclusive_transaction` claim with its
`OperationalError` "is locked" branch, `run_task`, the `batch` and `max_tasks` early returns,
`close_old_connections()` and the interval sleep — with exactly one line inserted:

```python
            close_old_connections()

            touch_heartbeat(config.WORKER_HEARTBEAT_PATH)   # <-- the only addition

            if self.running and not task_result:
                time.sleep(self.interval)
```

The position is the whole point: after `close_old_connections()` and before the interval sleep, so
it runs on every iteration whether or not a task was claimed. Nowhere else — not on a timer, not
inside `run_task()`, or a worker wedged inside a task stays green.

Three things about the copy that are easy to lose:

- Upstream's startup-delay line is `time.sleep(random.random())  # noqa: S311`. `S` is a selected
  ruff rule set here, so an implementer who reconstructs the loop from prose rather than copying it
  will drop the `noqa` and fail lint. Copy the line whole.
- The copied body calls `logger.info(...)` and `logger.warning(...)`. Bind
  `logger = logging.getLogger("django_tasks_db")` at the top of `worker.py`, the same name upstream
  uses, so those messages stay on the channel anyone already watching a `db_worker` is watching.
  This is not new logging — it is the copied lines keeping their existing destination.
- The per-iteration guarantee holds only for `batch=False` and `max_tasks=None`: upstream's two
  early `return None` branches sit *before* `close_old_connections()`, so a batch worker would stop
  touching the heartbeat the moment its queue drained. `fls_run_worker` hardcodes both values, so it
  holds in practice — say so in `HeartbeatWorker`'s docstring, since the class does not enforce it.

### `fls_run_worker.py`

Plain `BaseCommand`, no arguments, no `--reload`:

```python
class Command(BaseCommand):
    help = (
        "Run the database task worker with a heartbeat file and a watchdog that "
        "exits the process when the work loop stalls. Anyone needing the full flag "
        "surface runs db_worker."
    )

    def handle(self, *args: object, **options: object) -> None:
        heartbeat_path = config.WORKER_HEARTBEAT_PATH
        touch_heartbeat(heartbeat_path)
        start_watchdog(heartbeat_path, config.WORKER_HEARTBEAT_MAX_AGE_SECONDS)

        worker = HeartbeatWorker(
            queue_names=["*"],          # every queue; db_worker's own default is "default" alone
            interval=1,
            batch=False,
            backend_name=DEFAULT_TASK_BACKEND_ALIAS,
            startup_delay=True,
            max_tasks=None,
            worker_id=get_random_id(),
        )
        worker.configure_signals()
        worker.run()
```

`DEFAULT_TASK_BACKEND_ALIAS` comes from `django_tasks`, `get_random_id` from `django_tasks.utils` —
the same sources `db_worker` uses, so the defaults track upstream rather than being retyped.

### Tests (write first)

`freedom_ls/deployment/tests/test_worker.py`. Test the heartbeat and watchdog as directly callable
functions. Do **not** start a real loop: no FLS convention does, the suite runs on
`ImmediateBackend`, and `--disable-socket` is in effect.

**Two rules that hold for every test in this file and in `test_housekeeping.py`.** Both are there to
stop this feature's own test suite from arming the trap the feature exists to fire.

1. **Never let the default heartbeat path reach the filesystem.** `WORKER_HEARTBEAT_PATH` and
   `HOUSEKEEPING_HEARTBEAT_PATH` both default to the literal `/tmp/heartbeat`. A test that runs
   `handle()` without overriding it writes a real file on the developer's machine, outside
   `tmp_path`, that nothing cleans up — and because the two settings share a default, a worker test
   and a housekeeping test would write the same file and read each other's mtime. Put an `autouse`
   fixture at the top of each file that overrides the app's heartbeat setting to
   `str(tmp_path / "heartbeat")` for every test in it.
2. **Never leave a real watchdog thread running.** `Command.handle` starts a daemon thread polling
   every `WATCHDOG_POLL_SECONDS` against a 300-second staleness window. If the rest of the suite
   runs longer than that window — which the full FLS suite plausibly does — that thread finds the
   heartbeat stale and calls the real `os._exit(1)`, killing the pytest process mid-run with no
   traceback and no test failure to explain it. Any test that calls `call_command("fls_run_worker")`
   must patch `freedom_ls.deployment.management.commands.fls_run_worker.start_watchdog`.

- `touch_heartbeat` on an existing file advances its mtime.

- `django_tasks_db.__version__ == "0.12.0"`, with an assertion message naming

### `freedom_ls/deployment/housekeeping.py`

Import paths worth fixing here, because the obvious guesses are wrong: `get_date_max` is **not**
re-exported from `django_tasks_db` (that package's `__all__` is `["DatabaseBackend"]`), so it comes
`from django_tasks_db.models import DBTaskResult, get_date_max`. `TaskResultStatus` is reachable from
three places; use `from django_tasks import TaskResultStatus`, the top-level package, matching where
`fls_run_worker.py` takes `DEFAULT_TASK_BACKEND_ALIAS` from.

```python
def find_late_unpicked_tasks(max_age_seconds: int) -> QuerySet[DBTaskResult]:
    """READY task results that were due more than max_age_seconds ago.

    run_after is non-nullable and carries the sentinel get_date_max() (year 9999)
    for a task enqueued with no schedule, so lateness is measured from enqueued_at
    on that branch and from run_after on the other. A future run_after never
    satisfies run_after__lte=cutoff, so "not yet due is not late" needs no special
    case. Every queue, no scoping.
    """
    cutoff = timezone.now() - timedelta(seconds=max_age_seconds)
    return DBTaskResult.objects.filter(status=TaskResultStatus.READY).filter(
        Q(run_after=get_date_max(), enqueued_at__lte=cutoff)
        | Q(run_after__lt=get_date_max(), run_after__lte=cutoff)
    )


# The two ways a sweep can fail without being a bug in this command: the command
# it delegates to reports failure, or the database refuses the work. Anything else
# is a bug and propagates, where the management-command machinery turns it into a
# traceback and Sentry's own excepthook integration reports it.
SWEEP_FAILURES = (CommandError, DatabaseError)


def run_housekeeping_sweeps() -> list[str]:
    """Run both sweeps and the late-task check. Return a list of failure messages.

    Each step gets its own try/except so no failure stops a later step: a prune that
    dies must still leave expired sessions cleared, and both must still leave the
    late-task check run.
    """
    failures: list[str] = []

    try:
        call_command("prune_db_task_results", queue_name="*")
    except SWEEP_FAILURES as exc:
        sentry_sdk.capture_exception(exc)
        failures.append(f"prune_db_task_results failed: {exc}")

    try:
        call_command("clearsessions")
    except SWEEP_FAILURES as exc:
        sentry_sdk.capture_exception(exc)
        failures.append(f"clearsessions failed: {exc}")

    try:
        late_count = find_late_unpicked_tasks(
            config.HOUSEKEEPING_UNPICKED_TASK_MAX_AGE_SECONDS
        ).count()
    except DatabaseError as exc:
        sentry_sdk.capture_exception(exc)
        failures.append(f"late unpicked task check failed: {exc}")
    else:
        if late_count:
            sentry_sdk.capture_message(
                f"{late_count} READY task result(s) unpicked for more than "
                f"{config.HOUSEKEEPING_UNPICKED_TASK_MAX_AGE_SECONDS}s; "
                f"no worker is consuming."
            )
            failures.append(f"{late_count} task result(s) unpicked past the window.")

    return failures
```

`call_command` does not catch `CommandError` — that lives in `run_from_argv` — so a failing sweep
propagates into the local `except`. `--backend` needs no value: FLS has one task backend, aliased
`default`.

**Why these exception types and not `Exception`.** CLAUDE.md requires specific exception types, and
the two named here are the two ways a sweep fails without the fault being in this command: the
delegated command reports failure (`CommandError`), or the database refuses the work
(`DatabaseError`, which is the parent of `OperationalError` and `ProgrammingError`). Both are needed
— catching only `CommandError` would let a `DatabaseError` inside `.delete()` or `clear_expired()`
abort the whole command and skip the later steps, which is the behaviour the spec's per-step
`try`/`except` exists to prevent. Anything outside those two is a bug in this code, and it should
surface as a traceback and a non-zero exit rather than be filed as
`"clearsessions failed: <str(exc)>"`. That is not a silent failure: Sentry's Django integration
installs an excepthook, so an unanticipated exception reaching the top of the process is still
reported. The narrow catch follows the precedent at `freedom_ls/reports/tasks.py:57-69`. Sentry's Django integration never sees any of these, because the `except` blocks swallow
them by design; `init_sentry()` is a no-op without a DSN, so the calls are safe in dev and CI.

### `fls_run_housekeeping.py`

Plain `BaseCommand`, no arguments. Runs once and exits; the downstream wrapper supplies the loop and
the interval.

```python
class Command(BaseCommand):
    help = (
        "Prune finished task results, clear expired sessions, and report task "
        "results left unpicked past the configured window. Runs once and exits."
    )

    def handle(self, *args: object, **options: object) -> None:
        failures = run_housekeeping_sweeps()
        if failures:
            raise CommandError("; ".join(failures))
        # Only a fully clean run counts as a heartbeat: both sweeps succeeded and nothing
        # was left unpicked.
        touch_heartbeat(config.HOUSEKEEPING_HEARTBEAT_PATH)
        self.stdout.write("Housekeeping complete.")
```

`touch_heartbeat` is imported from `freedom_ls.deployment.worker` — one implementation, not two. It
stays in `worker.py` rather than moving to a module of its own: the spec places it there, and a
third module would be a name this feature invents. `CommandError` gives `run_from_argv` a message on
stderr and a non-zero exit.

### Tests (write first)

`freedom_ls/deployment/tests/test_housekeeping.py`. Use `stdout=StringIO()` on `call_command` and

**The prune backend problem.** Under the test settings `TASKS["default"]` is `ImmediateBackend`, so
`call_command("prune_db_task_results", ...)` raises `CommandError` unconditionally. Every test that
needs a *clean* run must therefore wrap in
`override_settings(TASKS={"default": {"BACKEND": "django_tasks_db.DatabaseBackend"}})`. Establish
this as a fixture at the top of the file and use it everywhere except the deliberately-broken-prune
tests.

That override does reset the cached backend handler — no workaround needed. `django_tasks_db` is
always in `INSTALLED_APPS`, its `backend.py` imports `django_tasks.signals` at module level, and that
import registers `clear_tasks_handlers`, a `setting_changed` receiver that clears
`django_tasks.task_backends` whenever `TASKS` changes. That is the same singleton
`prune_db_task_results`' `valid_backend_name()` reads.

Cases:

- A prune that raises still runs the session sweep, and vice versa (patch
  `freedom_ls.deployment.housekeeping.call_command` with a `side_effect` keyed on the command name;
  assert the other sweep still ran). Raise a `CommandError` from the side effect — an exception
  outside `SWEEP_FAILURES` is meant to propagate, so raising one here would test the opposite of
  what this case is for.
- An exception outside `SWEEP_FAILURES` propagates rather than being collected: patch the prune to
  raise a `ValueError` and assert `pytest.raises(ValueError)`, not `CommandError`. This is the guard
  on the narrowed catch — without it, someone widening it back to `Exception` breaks nothing.
- Both failures reach Sentry: `sentry_sdk.capture_exception` called once per failing step.
- The command exits non-zero — `pytest.raises(CommandError)` — in either case, and the heartbeat
  file is not created (`tmp_path` heartbeat + `override_settings(HOUSEKEEPING_HEARTBEAT_PATH=...)`).
- A clean run touches the heartbeat.
- Lateness, via `find_late_unpicked_tasks` directly against real `DBTaskResult` rows. `enqueued_at`
  is `auto_now_add`, so create the row under `time_machine.travel(...)` at the age you want rather

  follow-up `DBTaskResult.objects.filter(pk=...).update(enqueued_at=...)`. Cases:
  - `READY`, `run_after=get_date_max()`, enqueued two hours ago → late.
  - `READY`, `run_after` an hour in the future → not late.
  - `READY`, `run_after` two hours ago → late.
  - `SUCCESSFUL` of any age → not late.
- Late tasks reach `sentry_sdk.capture_message` and make the command exit non-zero.
- Expired sessions go and live ones stay: create one `Session` with `expire_date` in the past and
  one in the future, run the command, assert only the live one remains.
