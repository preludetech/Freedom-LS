## Shared design: `SittingPath` and the evaluator

Everything below the importer reads the path through one object. It is defined in Slice 3 and
restated here so every later slice uses the same names.

```python
# freedom_ls/form_engine/conditions.py
@dataclass(frozen=True, eq=False)
class SittingPath:
    """The pages, questions and content blocks one sitting is asked, in authored order.

    Page numbers are authored indexes (the URL's numbers); positions are places
    among the on-path pages (what the learner reads). The two only coincide
    when nothing is skipped.
    """

    all_pages: tuple[FormPage, ...]                      # every page, authored order
    children_by_page: Mapping[UUID, tuple[FormContent | FormQuestion, ...]]  # on-path page id -> its on-path children, page order
    question_ids: frozenset[UUID]                        # on-path questions
    content_ids: frozenset[UUID]                         # on-path content blocks

    numbered_pages() -> list[tuple[int, FormPage]]       # (authored number, page) for each on-path page, in order
    pages -> list[FormPage]                              # the pages of numbered_pages()
    page_count -> int
    has_page_number(number) -> bool
    has_question(question_id) -> bool
    children_of(page) -> list[FormContent | FormQuestion]
    questions_of(page) -> list[FormQuestion]
    questions() -> list[FormQuestion]                    # every on-path question, sitting order
    page_position(number) -> int                         # 1-based place among on-path pages
    question_positions() -> dict[UUID, int]              # 1-based place among on-path questions, built once
    next_page_number(after: int) -> int | None           # first on-path page after `after`
    previous_page_number(before: int) -> int | None
    first_page_number_at_or_after(number) -> int | None
    last_page_number() -> int | None                     # None only for a zero-page form
    landing_page_number(number) -> int | None            # off-path request: next on-path after it, else last
    as_asked() -> dict[str, list[str]]                   # {"pages": [...], "questions": [...], "content": [...]}
```

`page_ids` is `children_by_page.keys()`. The two position lookups are dicts computed once in
`__post_init__` (set with `object.__setattr__`, since the dataclass is frozen), so a message builder
that numbers several questions pays for one walk.

> **Assumed default:** `SittingPath` keeps the wide method surface above rather than a minimal core
> composed at call sites. Why: every method has at least one caller in a later slice, and one
> vocabulary for "next on-path page" or "position" beats each shell spelling it out. To change: trim
> the list in 3a and inline the dropped helpers where they are used.

Page 1 can never carry a condition (its source would have to be on an earlier page), so a form with
at least one page always has at least one on-path page.

```python
FORM_TREE_LOOKUPS = (
    "pages__condition_option",
    "pages__questions__condition_option",
    "pages__text_items__condition_option",
)

def load_form_tree(form: Form) -> None:
    """Prefetch the pages, their children and every condition_option.

    Plain lookup strings rather than Prefetch objects with select_related, so
    a form whose pages were already prefetched by a plainer lookup still gets
    the condition_option level fetched instead of skipped.
    """
    prefetch_related_objects([form], *FORM_TREE_LOOKUPS)

def answers_by_question(form_progress: FormProgress) -> dict[UUID, QuestionAnswer]:
    """The sitting's answers keyed by question id, with selected options loaded.

    Always read fresh: a page submission changes the answers mid-request, and
    the path has to follow them.
    """
    answers = QuestionAnswer.objects.filter(form_progress=form_progress).prefetch_related("selected_options")
    return {answer.question_id: answer for answer in answers}

def _condition_holds(option: QuestionOption | None, selected: Mapping[UUID, frozenset[UUID]]) -> bool:
    # `selected` holds only on-path questions, so an unanswered or off-path
    # source makes the condition false.
    return option is None or option.id in selected.get(option.question_id, frozenset())

def _on_path_children(page: FormPage, selected: Mapping[UUID, frozenset[UUID]]) -> list[FormContent | FormQuestion]:
    return [child for child in page.children() if _condition_holds(child.condition_option, selected)]

def compute_path(form: Form, answers: Mapping[UUID, QuestionAnswer]) -> SittingPath:
    load_form_tree(form)
    option_ids = {qid: frozenset(o.id for o in answer.selected_options.all()) for qid, answer in answers.items()}
    selected: dict[UUID, frozenset[UUID]] = {}          # grows as questions prove on-path
    children_by_page: dict[UUID, tuple[FormContent | FormQuestion, ...]] = {}
    all_pages = tuple(form.pages.all())
    for page in all_pages:
        if not _condition_holds(page.condition_option, selected):
            continue
        children = _on_path_children(page, selected)
        if page.children() and not children:
            continue
        children_by_page[page.id] = tuple(children)
        for child in children:
            if child.content_type == "FORM_QUESTION":
                selected[child.id] = option_ids.get(child.id, frozenset())
    return _path(all_pages, children_by_page)

def _path(all_pages, children_by_page) -> SittingPath:
    """Derive question_ids and content_ids from the children map."""

def current_path(form_progress: FormProgress) -> SittingPath:
    """The path the sitting's answers give right now."""
    return compute_path(form_progress.form, answers_by_question(form_progress))

def frozen_path(form: Form, asked: Mapping[str, list[str]]) -> SittingPath:
    """The path a completed sitting recorded, rebuilt over the current tree."""
    load_form_tree(form)
    page_ids = {UUID(value) for value in asked["pages"]}
    child_ids = {UUID(value) for key in ("questions", "content") for value in asked[key]}
    children_by_page = {page.id: tuple(c for c in page.children() if c.id in child_ids) for page in form.pages.all() if page.id in page_ids}
    return _path(tuple(form.pages.all()), children_by_page)

def full_path(form: Form) -> SittingPath:
    """Every page and child: what a sitting completed before paths were recorded was asked."""
    load_form_tree(form)
    return _path(tuple(form.pages.all()), {page.id: tuple(page.children()) for page in form.pages.all()})

def sitting_path(form_progress: FormProgress, answers: Mapping[UUID, QuestionAnswer] | None = None) -> SittingPath:
    """Frozen when the sitting recorded one, everything for an older completed
    sitting, else computed from `answers` (read fresh when not given)."""
    if form_progress.asked is not None:
        return frozen_path(form_progress.form, form_progress.asked)
    if form_progress.completed_time is not None:
        return full_path(form_progress.form)
    if answers is None:
        answers = answers_by_question(form_progress)
    return compute_path(form_progress.form, answers)
```

`page.children()` reads `text_items.all()` and `questions.all()`, so after `load_form_tree` it costs
no query. `compute_path` is six queries for the tree (three levels, each with its option) plus two
for `answers_by_question`, and fewer when a caller's prefetch already covered some levels, because
`prefetch_related_objects` skips a level it finds fetched and carries on below it. Every request
computes its path once, after any save it computes it again, and the query-budget tests bound the
total.

## Slice 1: an author can write `slug` and `when`, and the importer stores them

**Behaviour.** A form directory with `slug` on questions and `when` on pages, questions or content
imports; each gated row's `condition_option` points at the `QuestionOption` of this import run that
the condition names; removing `when` clears it; a form with neither key imports exactly as before.
Nothing reads the condition yet.

### 1a. Schema (`freedom_ls/form_engine/schema.py`)

```python
class Condition(BaseModel):
    """`when:` on a page, question or content block."""

    model_config = ConfigDict(extra="forbid")

    question: str = Field(..., description="The slug of the source question, on an earlier page of this form")
    equals: str | None = Field(None, description="For a multiple_choice or dropdown source: the option value that must be selected")
    includes: str | None = Field(None, description="For a checkboxes source: an option value that must be among those selected")

    @field_validator("equals", "includes", mode="before")
    @classmethod
    def _literal_as_str(cls, value: object) -> str | None:
        # yaml.safe_load reads an unquoted yes/no/on/off as a bool, so a bool here
        # is almost always an author who meant the word. The option values it is
        # compared against are strings (QuestionOption.value), so an int is one too.
        if isinstance(value, bool):
            raise ValueError(f"{value} was read as a boolean. Quote it (\"{str(value).lower()}\") so it is compared as text.")
        if value is None:
            return None
        return str(value)

    @model_validator(mode="after")
    def _exactly_one_test(self) -> "Condition":
        if (self.equals is None) == (self.includes is None):
            raise ValueError("when needs exactly one of equals or includes.")
        return self

    @property
    def literal(self) -> str:
        """The option value the condition tests, whichever key carried it."""
        return self.equals if self.equals is not None else self.includes
```

The bool message quotes the lower-cased spelling (`"true"`/`"false"`) because the parser has already
lost whether the author wrote `yes` or `true`; the hint is "quote it", and that is what matters.

- `FormQuestion` gains `slug: str | None = Field(None, pattern=r"^[a-z0-9_]+$", max_length=100, description=...)`,
  the same length as the model field.
- `FormPage`, `FormQuestion` and `FormContent` gain `when: Condition | None = Field(None, ...)`.

  ```python
  def _resolve_condition(
      when: Condition | None,
      options_by_slug: Mapping[str, Mapping[str, QuestionOption]],
      file_path: Path,
  ) -> QuestionOption | None:
      """The option this run saved for the value `when` names, or None.

      Resolved from this run's saved objects rather than the database, where
      questions dropped from the YAML and options re-created without a uuid
      leave stale rows that a value lookup could hit.
      """
      if when is None:
          return None
      option = options_by_slug.get(when.question, {}).get(when.literal)
      if option is None:
          raise ValueError(f"{file_path}: when names {when.question} = {when.literal}, which no earlier question in this form saved.")
      return option
  ```

  importer's only guard.

Run it in `validate()` beside `validate_category_references`. Algorithm:

```python
def validate_form_conditions(all_parsed: list[BaseBaseContentModel]) -> list[str]:
    """Check every `when` against the questions the same form declares earlier.

    Page order is the one content_save assigns: the page files sorted as
    paths, so `10. x.yaml` comes before `2. y.yaml`. Every error is reported,
    never just the first.
    """
    errors: list[str] = []
    forms = {item.file_path.parent: item for item in all_parsed if item.content_type == ContentType.FORM}
    sections_by_file = group FORM_PAGE / FORM_QUESTION / FORM_CONTENT items by file_path, file order kept
    for form_dir, form in forms.items():
        page_files = sorted(f for f in sections_by_file if f.parent == form_dir)
        gated = [(index, item) for index, f in enumerate(page_files) for item in sections_by_file[f] if item.when is not None]
        if form.strategy == FormStrategy.QUIZ:
            if gated: errors.append(error 7, once, naming the first gated file and target)
            continue                                   # a QUIZ form never reaches rules 2-6
        slug_owner, duplicates = _slug_owners(page_files, sections_by_file)   # slug -> (page index, question item)
        errors.extend(duplicates)                                             # rule 1, naming both files
        for page_index, item in gated:
            problem = _condition_problem(item.when, page_index, slug_owner)   # rules 2-6, first that applies, or None
            if problem is not None:
                errors.append(_condition_error(item.file_path, _target_label(item), *problem))
    return errors
```

`_condition_problem` returns `(problem, fix)` for the first of rules 2–6 that applies: unknown slug
(listing `sorted(slug_owner)`), source on this page or later (`source_index >= page_index`), source
type outside `QuestionType.MULTIPLE_CHOICE` / `DROPDOWN` / `CHECKBOXES`, `equals` on a `CHECKBOXES`
source or `includes` on another, literal not among `str(option.value)` for the source's options
(listing them). `_target_label` reads `page "About you"`, `question "What is ..."` or
`content block 2`. The sort is over the same `Path` objects `content_save` sorts (its tuples sort
on the path first and never tie), and the `10.` versus `2.` test guards the agreement.

Each error goes through a local `_condition_error(file_path, target, problem, fix)` shaped like
`_reference_error`: header `Invalid condition`, the file, `<target> → when`, the problem, the fix.
Messages, in the voice of `validate_quiz_fields`:

1. `slug "employer_pays" is declared twice: in <file A> and <file B>. Give each question its own slug.`
2. `when.question names "employer_pay", which no question in this form has as its slug. Declared slugs: employer_pays, dietary_needs.`
3. `when.question names "employer_pays", which is on this page or a later one. A condition's source has to be on an earlier page.`
4. `when.question names "full_name", a short_text question. A source has to be a multiple_choice, dropdown or checkboxes question.`
5. `when.equals cannot test "dietary_needs", a checkboxes question. Use includes.` / `when.includes cannot test "employer_pays", a multiple_choice question. Use equals.`
6. `when.equals names "yse", which is not a value of "employer_pays". Its values: yes, no.`
7. `when is not allowed in a QUIZ form. Remove it, or change the form's strategy.`

Rule 3 covers cycles: a source on an earlier page cannot depend on a later one.

### 5a. `FormProgress.get_current_page_number(path, answered_ids)` (`models.py`)

```python
def get_current_page_number(self, path: SittingPath, answered_ids: set[UUID]) -> int:
    """The first on-path page with an on-path question that has no answer,
    else the last on-path page (1 when the form has no pages)."""
    for number, page in path.numbered_pages():
        if any(q.id not in answered_ids for q in path.questions_of(page)):
            return number
    return path.last_page_number() or 1
```

### 5b. `paging.py`

```python
def _answered_ids(form_progress: FormProgress) -> set[UUID]:
    return set(form_progress.answers.values_list("question_id", flat=True))

def _furthest_answered_page(path: SittingPath, answered_ids: set[UUID]) -> int:
    """The highest on-path page number carrying an answer to an on-path question, or 0."""

def _resume_page_number(form_progress: FormProgress, path: SittingPath) -> int:
    numbered = path.numbered_pages()
    if not numbered:
        return 1
    answered_ids = _answered_ids(form_progress)
    candidate = min(max(form_progress.get_current_page_number(path, answered_ids),

    landed = path.first_page_number_at_or_after(candidate)
```

### 6a. `FormProgress.complete()` and `score(path)` (`models.py`)

```python
def complete(self) -> None:
    """Finish the sitting: keep only the answers it was asked, record what it
    was asked, score it, then tell the apps above (idempotent)."""
    if self.completed_time:
        return
    # Local import: conditions imports this module.
    from .conditions import current_path

    with transaction.atomic():
        path = current_path(self)
        # Deleting cascades to any attached file row; its receiver removes the
        # stored object only once this block commits.
        self.answers.exclude(question_id__in=path.question_ids).delete()
        self.asked = path.as_asked()
        self.completed_time = timezone.now()
        self.score()
        self.save()
    # Sent once the block has committed or released its savepoint, not through
    # on_commit: the receivers that update course progress and webhooks read
    # the row this block wrote.
    form_attempt_completed.send(sender=type(self), user=self.user, form=self.form, attempt=self)
```

from `complete()`.

```python
            # The edit opened a page the applicant has never filled in; walk
            # them through it before sending them back to review.
            return redirect(f"{url_for_page(onward)}?return={RETURN_TO_CHECK}")
        return redirect("course_applications:check_answers", pk=app.pk)
    next_number = path.next_page_number(page_number)
    ...
```

`url_for_page` stays the plain page URL; the marker is appended here only.
