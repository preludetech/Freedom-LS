### 2a. `can` (`freedom_ls/learner_management/capabilities.py`, new)

It has private builders, one per resolution step. `queries.py` reuses them in slice 4, and slice 5
adds the `OrganisationMember` gate inside them.

```python
if TYPE_CHECKING:
    type RequestUser = User | AnonymousUser | AbstractBaseUser

def roles_granting(capability: str, site: Site) -> frozenset[str]:
    config = get_role_config(site.name)
    return frozenset(key for key, role in config.items() if capability in role.permissions)

def _site_of(scope: Model) -> Site:            # a Site is its own site
def _organisation_of(scope: Model) -> Organisation | None:
    # Site → None; Organisation → itself; Cohort / Learner → .organisation;
    # anything else → raise TypeError (no silent deny for an unsupported scope object)

def _site_grants(user, roles, site) -> QuerySet[SiteRoleAssignment]:          # step 3
    SiteRoleAssignment.objects.filter(user=user, site=site, role__in=roles, is_active=True)

def _active_role_assignments(model, roles) -> QuerySet[ObjectRoleAssignment]:
    # the one place the assignment-table join is written
    return ObjectRoleAssignment.objects.filter(
        role__in=roles, is_active=True, content_type=ContentType.objects.get_for_model(model))

def _granted(model, user, roles) -> QuerySet:
    return model.objects.filter(Exists(_active_role_assignments(model, roles).filter(
        user=user, object_id=Cast(OuterRef("pk"), CharField()))))

def _granted_organisations(user, roles) -> QuerySet[Organisation]:            # step 5
    return _granted(Organisation, user, roles)

def _granted_cohorts(user, roles) -> QuerySet[Cohort]:                        # step 6
    return _granted(Cohort, user, roles)

def can(user: RequestUser, capability: str, scope: Model) -> bool:
    if not user.is_authenticated or not user.is_active: return False
    user = cast("User", user)
    if user.is_superuser: return True
    site = _site_of(scope)
    roles = roles_granting(capability, site)
    if not roles: return False
    if _site_grants(user, roles, site).exists(): return True
    organisation = _organisation_of(scope)
    if organisation is None: return False
    if _granted_organisations(user, roles).filter(pk=organisation.pk).exists(): return True
    cohorts = _granted_cohorts(user, roles)
    if isinstance(scope, Cohort): return cohorts.filter(pk=scope.pk).exists()
    if isinstance(scope, Learner):
        return CohortMembership.objects.filter(learner=scope, cohort__in=cohorts).exists()
    return False
```

`ObjectRoleAssignment.object_id` is a `CharField` and the targets have UUID keys, so the join
casts `OuterRef("pk")`. It stays in the ORM. `can` ignores the scope object's own `is_active`.

  VIEW_COHORT = "freedom_ls_learner_management.view_cohort"
  VIEW_LEARNER = "freedom_ls_learner_management.view_learner"

  def organisations_accessible_to(user):
      if not active: none();  if superuser: all()
      site = _current_site()

          | Q(pk__in=_granted_cohorts(user, roles_granting(VIEW_COHORT, site)).values("organisation_id"))

  def cohorts_visible_to(user, organisation):
      within = Cohort.objects.filter(organisation=organisation)
      roles = roles_granting(VIEW_COHORT, organisation.site)
      if superuser or _site_grants(...).exists() or _granted_organisations(user, roles).filter(pk=organisation.pk).exists():

## Slice 10: checked assignment utilities

**Behaviour:** `assign_role`, `remove_role` and `set_organisation_member_active` enforce who may
assign what, self-changes, inactive targets and the last organisation admin. Spec 9 builds the
screens on them.

- `freedom_ls/learner_management/role_assignment.py` (new):

  ```python
  ASSIGN = "freedom_ls_role_based_permissions.assign_{role}"
  CHANGE_MEMBER = "freedom_ls_learner_management.change_organisationmember"

  def _refuse_unless(grantor, user, capability, target, *, assigning: bool) -> None:
      if grantor == user: raise RoleChangeRefused(RefusalReason.SELF)
      if assigning and not user.is_active: raise RoleChangeRefused(RefusalReason.INACTIVE_USER)
      if not can(grantor, capability, target): raise RoleChangeRefused(RefusalReason.NOT_PERMITTED)

  def _is_site_level_admin(user, site) -> bool:   # superuser, or an active site_admin assignment on the site

  def _holds_organisation_admin(user, organisation) -> bool:
      # an active organisation_admin ObjectRoleAssignment on it, via _active_role_assignments

  def _lock_and_refuse_last_organisation_admin(organisation, user) -> None:
      # caller is inside transaction.atomic()
      Organisation.objects.select_for_update().get(pk=organisation.pk)
      if not active_organisation_admins(organisation).exclude(pk=user.pk).exists():
          raise RoleChangeRefused(RefusalReason.LAST_ORGANISATION_ADMIN)

  def assign_role(grantor, user, role, target) -> None:
      _refuse_unless(grantor, user, ASSIGN.format(role=role), target, assigning=True)
      if isinstance(target, Site): assign_site_role(user, role, site=target, assigned_by=grantor)
      else: assign_object_role(user, target, role, assigned_by=grantor)

  def remove_role(grantor, user, role, target) -> None:
      _refuse_unless(grantor, user, ASSIGN.format(role=role), target, assigning=False)
      if isinstance(target, Site):
          remove_site_role(user, role, site=target, removed_by=grantor)   # LAST_SITE_ADMIN lives there
          return
      with transaction.atomic():
          if role == "organisation_admin" and _holds_organisation_admin(user, target) and not _is_site_level_admin(grantor, target.site):
              _lock_and_refuse_last_organisation_admin(target, user)
          remove_object_role(user, target, role, removed_by=grantor)

  def set_organisation_member_active(actor, member, is_active) -> None:
      _refuse_unless(actor, member.user, CHANGE_MEMBER, member.organisation, assigning=is_active)
      with transaction.atomic():
          if not is_active and _holds_organisation_admin(member.user, member.organisation) and not _is_site_level_admin(actor, member.site):
              _lock_and_refuse_last_organisation_admin(member.organisation, member.user)
          member.is_active = is_active
          member.save(update_fields=["is_active"])
  ```

  Type the arguments as the spec gives them. `_refuse_unless` runs the spec's checks in order:
  `SELF`, then `INACTIVE_USER`, then `NOT_PERMITTED`. A downstream role with no `assign_` string is
  therefore refused as `NOT_PERMITTED`. The receiver creates the `OrganisationMember` for a new
  grant and never reactivates an inactive one.

**Tests:** `freedom_ls/learner_management/tests/test_role_assignment.py` (new):
- one test per refusal reason, each asserting that nothing changed;
- an organisation admin assigning in another organisation → `NOT_PERMITTED`;
- a cohort admin assigning anything → `NOT_PERMITTED`;
- a site admin removing the last organisation admin succeeds;
- the last organisation admin removing themselves → `SELF`;
- deactivating the last organisation admin's `OrganisationMember` → `LAST_ORGANISATION_ADMIN`;
- assigning to a user with an inactive `OrganisationMember` leaves it inactive;
- removing `site_admin` through `remove_role` surfaces `LAST_SITE_ADMIN`;
- under `CaptureQueriesContext`, an organisation-admin removal's SQL contains `FOR UPDATE` on the
  `freedom_ls_organisations_organisation` row.
