  def _resolve_course_organisations(item, site) -> list[Organisation]:
      if not item.organisations:
          return []
      found = list(Organisation._base_manager.filter(site=site, slug__in=item.organisations))
      unknown = sorted(set(item.organisations) - {o.slug for o in found})
      if unknown:

          raise ValueError(

              f"Valid organisation slugs: {valid}"

### Slice 2: the audience gate

**Behaviour.** A learner who is not in the audience of an organisation course does not see it on
the dashboard, the all-courses list or recommendations, gets a 404 on its detail, home, access and
course item URLs, and cannot embed it with the markdown course card. A registered learner keeps
access.

**Files.**

- `freedom_ls/learner_management/queries.py`, beside `is_registered_for_course_expression`:

  ```python
  def is_organisation_course_expression() -> Exists:
      # The through table rather than a join on `organisations`, so the
      # predicate adds no rows to the course queryset it is embedded in.
      return Exists(Course.organisations.through.objects.filter(course_id=OuterRef("pk")))


  def _open_to_everyone_expression() -> Q:
      # The half both audience rules share: a course that belongs to no
      # organisation, or one that is public, is open whoever is asking.
      return ~is_organisation_course_expression() | Q(audience=CourseAudience.PUBLIC)


  def in_audience_expression(user: RequestUser) -> Q:
      """Courses this user may see: every course open to everyone, and for a
      signed-in user the courses of the organisations they hold an active
      Learner row in, plus any course they are registered for, so a change to
      a course's organisations never strands a learner mid-course."""
      if not user.is_authenticated:
          return _open_to_everyone_expression()
      own_organisation_learner = Exists(
          Learner.objects.filter(
              user=user, is_active=True, organisation__courses=OuterRef("pk")
          )
      )
      return (
          _open_to_everyone_expression()
          | own_organisation_learner
          | is_registered_for_course_expression(user)
      )


  def _course_matches(course: Course, expression: Q) -> bool:
      # _base_manager, not objects: the course was already fetched through the
      # site-aware manager and the filter is on its pk alone, so no row from
      # another site can match. The base manager lets the check run where no
      # request has pinned a site: the admin's clean(), a background rule run.
      return Course._base_manager.filter(pk=course.pk).filter(expression).exists()


  def is_in_audience(user: RequestUser, course: Course) -> bool:
      """Single-course form of in_audience_expression, evaluated through the
      same expression so the two cannot drift."""
      return _course_matches(course, in_audience_expression(user))
  ```

  > site, where `Course.objects` would fail open or closed unpredictably. **Assumed default:**
  > keep `_base_manager`, with the comment above and a test that a course of another site is
  > never reported in the audience. Why: the filter is on a pk already fetched site-aware, and
  > the `Learner` subquery stays on the site-aware manager. To change: this snippet.

### Slice 4: self-registration through the learner's own organisation

**Behaviour.** A learner who self-registers for an organisation course they hold an active
`Learner` row for lands on that organisation, so their in-progress card and player chip carry it.
A public organisation course with no such row still registers through the default organisation.
A learner the course is no longer open to through any organisation is refused.

  def own_organisation_for_course(user: User, course: Course) -> Organisation | None:
      """The earliest of the user's active Learner rows in one of the course's
      organisations, or None when they hold none."""
      learner = (
          Learner.objects.filter(user=user, is_active=True, organisation__courses=course)

          .order_by("created_at")

      )
