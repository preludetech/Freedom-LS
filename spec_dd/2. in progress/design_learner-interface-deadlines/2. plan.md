### 1b. `DeadlineTier` and `deadline_tier` (`freedom_ls/learner_interface/utils.py`)

```python
class DeadlineTier(StrEnum):
    UPCOMING = "UPCOMING"
    TODAY = "TODAY"
    MISSED = "MISSED"
    CLOSED = "CLOSED"


def deadline_tier(
    due_at: datetime,
    *,
    is_overdue: bool,
    is_closed: bool,
    now: datetime,
    zone: ZoneInfo,
) -> DeadlineTier:
    """How urgent an unfinished deadline looks.

    Closed outranks missed, and missed outranks today. "Today" is judged in the site
    zone, the zone an undated deadline's end of day is fixed in, so a learner elsewhere
    can read "Today" on a date their browser calls tomorrow. A past deadline whose
    policy does not mark it overdue is upcoming: that switch suppresses every late mark."""
    if is_closed:
        return DeadlineTier.CLOSED
    if is_overdue:
        return DeadlineTier.MISSED
    if now < due_at and due_at.astimezone(zone).date() == now.astimezone(zone).date():
        return DeadlineTier.TODAY
    return DeadlineTier.UPCOMING
```

### 1c. Shared deadline selection (`utils.py`)

Move `COURSE_DEADLINE_TITLE`, `CardDeadline`, `CardDeadlineLine` and `_card_deadline_line` from
`views.py` into `utils.py`, renaming the function `card_deadline_line`. `views.py` imports

```python
@dataclass(frozen=True)
class CardDeadline:
    """One deadline a card names: the item's title, when it is due and how urgent it looks."""

    title: str
    due_at: datetime
    is_overdue: bool
    status: str
    deadline_tier: DeadlineTier
    closes_at: datetime | None = None
    is_closed: bool = False
    # True for a deadline already behind the learner, whether or not it carries the
    # overdue mark; the eyebrow reads "Deadline" rather than "Next deadline" for it.
    is_past: bool = False


def _row_deadline(row: dict, *, is_past: bool) -> CardDeadline:
    """A card deadline read off one item row. A past row keeps its overdue mark and its
    closed state; a coming one has neither."""
    return CardDeadline(
        title=row["title"],
        due_at=row["due_at"],
        is_overdue=row["is_overdue"] if is_past else False,
        status=row["status"],
        deadline_tier=row["deadline_tier"],
        closes_at=row["closes_at"],
        is_closed=row["is_closed"] if is_past else False,
        is_past=is_past,
    )


def card_deadline_line(
    course_index: list[dict],
    course_deadlines: CourseDeadlines | None,
    deadline_policy: ResolvedDeadlinePolicy | None,
    now: datetime,
) -> CardDeadlineLine:
    # Only item rows carry due_at; a completed one is neither missed nor next. A past
    # row counts as missed whether or not it carries the overdue mark or has closed. An
    # in-progress course is never complete, so the time limit's overdue check takes
    # completed=False.
    open_rows = [
        row for row in _item_rows(course_index)
        if row.get("due_at") is not None and row["status"] != COMPLETE
    ]
    missed = [_row_deadline(row, is_past=True) for row in open_rows if row["due_at"] <= now]
    upcoming = [_row_deadline(row, is_past=False) for row in open_rows if row["due_at"] > now]
    if course_deadlines is not None and course_deadlines.time_limit.due_at is not None:
        time_limit = course_deadlines.time_limit
        overdue = deadline_policy is not None and shows_overdue(time_limit, deadline_policy, False, now)
        is_past = time_limit.due_at <= now
        # A time limit never closes an item, so its tier is never CLOSED.
        tier = deadline_tier(time_limit.due_at, is_overdue=overdue, is_closed=False,
                             now=now, zone=course_deadlines.zone)
        (missed if is_past else upcoming).append(
            CardDeadline(COURSE_DEADLINE_TITLE, time_limit.due_at, overdue, IN_PROGRESS, tier, is_past=is_past)
        )
    ...  # earliest() and access_ends_at unchanged, reading course_deadlines.time_limit or NO_DEADLINE
```

`_annotate_next_up` keeps its own two-level flatten for the next-up search and calls

### 2a. `site_zone_text` short form (`freedom_ls/site_aware_models/time_zones.py`)

```python
def site_zone_text(instant: datetime, *, short: bool = False) -> str:
    """... The short form is the day and abbreviated month, plus the year only when it is
    not this year, judged in the instant's own zone."""
    if short:
        text = f"{instant.day} {instant:%b}"
        if instant.year != timezone.now().astimezone(instant.tzinfo).year:
            text += f" {instant:%Y}"
        return text
    return f"{instant.day} {instant:%b %Y, %H:%M} {instant.tzname()}"
```

### 2b. `local_time` with `format` (`freedom_ls/deadlines/templatetags/deadline_tags.py`)

```python
@register.simple_tag(takes_context=True)
def local_time(context: Context, value: datetime, format: str = "") -> SafeString:
    """...
    "short" asks the browser for the day and month only; "today" is a fixed word, left
    alone by the browser because the server already decided it is today in the site zone."""
    instant = value.astimezone(UTC).isoformat(timespec="seconds")
    if format == "today":
        return format_html('<time datetime="{}">Today</time>', instant)
    show_zone = "true" if context.get("show_time_zone_to_learners") else "false"
    format_attr = format_html(' data-format="{}"', format) if format == "short" else ""
    return format_html(
        '<time datetime="{}" x-data="localTime" data-show-zone="{}"{}>{}</time>',
        instant, show_zone, format_attr, site_zone_text(value, short=format == "short"),
    )
```

The keyword is `format` because that is the name a template writes; the project's ruff rules do
not flag it.

### 2c. `localTime` (`freedom_ls/base/static/base/js/alpine-components.js`)

```js
Alpine.data("localTime", () => ({
    init() {
        const instant = new Date(this.$el.getAttribute("datetime"));
        if (Number.isNaN(instant.getTime())) return;
        const short = this.$el.dataset.format === "short";
        const options = short
            ? { day: "numeric", month: "short" }
            : { day: "numeric", month: "short", year: "numeric", hour: "2-digit", minute: "2-digit" };
        // A short date names its year only when it is not the viewer's current year.
        if (short && instant.getFullYear() !== new Date().getFullYear()) options.year = "numeric";
        if (!short && this.$el.dataset.showZone === "true") options.timeZoneName = "short";
        this.$el.textContent = new Intl.DateTimeFormat(undefined, options).format(instant);
    },
}));
```

Follow the CSP build rules in `.claude/ds/config.md` (`## Alpine.js`): server values come in
through `data-*` only, which this does.

### 4a. `views.py`

```python
@dataclass(frozen=True)
class PlayerCourseIndex:
    """The outline the player gates and chrome both read, with what it was resolved from, so
    the deadline the panel names is picked from the same rows and instants as the gates."""

    rows: list[dict]
    course_deadlines: CourseDeadlines | None
    deadline_policy: ResolvedDeadlinePolicy | None
    now: datetime

    @classmethod
    def empty(cls) -> PlayerCourseIndex:
        """An outline with no rows: what the chrome draws when the policy keeps nothing."""
        return cls(rows=[], course_deadlines=None, deadline_policy=None, now=timezone.now())


def _player_course_index(
    user: RequestUser,
    course: Course,
    index: int,
    course_progress: CourseProgress | None,
    *,
    access_ended_outline: bool = False,
) -> PlayerCourseIndex:
    now = timezone.now()
    course_deadlines = _deadlines_or_none(course_progress)
    deadline_policy = _deadline_policy_or_none(course_progress)
    rows = get_course_index(
        user=user, course=course, current_index=index, can_access_content=True,
        course_progress=course_progress, course_deadlines=course_deadlines,
        deadline_policy=deadline_policy, now=now, access_ended_outline=access_ended_outline,
    )
    return PlayerCourseIndex(rows, course_deadlines, deadline_policy, now)
```

`list[dict]` is the row type `get_course_index` already returns; keep it.

  read `player_index.rows`; pass `course_index=player_index` and
