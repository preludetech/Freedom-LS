Implements `1. spec.md`. Paths are relative to the repo root; `fd/` is `freedom_ls/`. Line
numbers are from the worktree on 2026-10-29 and drift as earlier slices land.

The spec's "Documentation and downstream" section (upgrade notes, product docs, the fls-content

## How to work through this plan

- TDD throughout: each slice names the behaviour and the tests that prove it. Write the failing
  test, then the smallest change that passes it, then the full suite before committing.
- Run tests the way `claude_plugins/django-stack/resources/test_tiers.md` says. Tests of the
  content validator run under the `dev_tooling` marker when the file beside them carries it
  (check `fd/content_engine/tests/test_validate.py` before adding to it).
- Every slice (or lettered sub-slice) leaves the suite green and is committed on its own with
  `uv run git commit`.
- Learner-facing copy is spelled exactly as the spec spells it. Comments and docstrings state the
  reason itself and never point at the spec, this plan, a bug id or a QA report (the
  `code-comments` skill).
- Where a slice adds Tailwind classes that no template uses yet, run `npm run tailwind_build`
  and check the class is generated.
- Test placement follows the mirroring rule: a test of `views.py` goes in `tests/test_views.py`,
  a test of a template with no module goes in an exempt directory (`tests/components/`) or a new
  subpackage listed in `test_organisation/mirroring_exemptions.txt` with a reason.

class DocumentTitleNode(template.Node):
    """The browser tab title: the page name, then the site name, or the site name alone."""

    def __init__(self, nodelist: template.NodeList) -> None:

```python
# Pages that never make sense to come back to: the auth pages themselves and
# the claim page, whose path is the credential.
_NO_RETURN_URL_NAME_PREFIX = "account_"
_NO_RETURN_VIEW_NAMES = frozenset({"course_applications:claim"})


def _request_next(context: template.Context) -> str | None:
    """The request's own safe `next`, or None."""
    request: HttpRequest | None = context.get("request")
    return get_next_redirect_url(request) if request is not None else None


@register.simple_tag(takes_context=True)
def url_back_here(context: template.Context, url_name: str) -> str:
    """Reverse `url_name` carrying a `next` that brings the visitor back to this page.

    A `next` already on the request wins. Otherwise the current path is used,
    except where returning makes no sense: the home page, the account pages,
    the claim page and the error pages.
    """
    request: HttpRequest | None = context.get("request")
    url = reverse(url_name)
    if request is None:
        return url
    if _request_next(context):
        return url_with_next(context, url_name)
    if request.method != "GET" or context.get("is_error_page"):
        return url
    if request.path == "/" or _is_no_return_path(request.path):
        return url
    back = request.get_full_path()
    allowed = url_has_allowed_host_and_scheme(
        back, allowed_hosts={request.get_host()}, require_https=request.is_secure()
    )
    if not allowed:
        return url
    return f"{url}?{urlencode({REDIRECT_FIELD_NAME: back})}"


def _is_no_return_path(path: str) -> bool:
    try:
        match = resolve(path)
    except Resolver404:
        return False
    url_name = match.url_name or ""
    return url_name.startswith(_NO_RETURN_URL_NAME_PREFIX) or match.view_name in _NO_RETURN_VIEW_NAMES
```

```python
from markdown.preprocessors import Preprocessor

# A component tag opening a line: `<c-accordion title="x">`. A self-closing tag
# (`<c-picture … />`) has no body and is left alone. The match stops at the
# first `>`, so an attribute value may not contain one.
_COMPONENT_OPEN_TAG = re.compile(r"^(?P<prefix>\s*<(?P<tag>c-[\w-]+)(\s[^>]*?)?)(?<!/)>")


class ComponentBodyPreprocessor(Preprocessor):
    """Mark every allowed component tag so md_in_html converts its body as Markdown.

    Runs after fenced_code has stashed code fences and before md_in_html reads
    the raw HTML, so an example inside a fence is never marked. Tags whose body
    is not Markdown (an equation, a code block) are skipped.

@register.simple_tag(takes_context=True)
def deferred_interest_course(context: template.Context) -> Course | None:
    """The coming-soon course a signup's `next` leads back to, if the visitor may see it."""
    request: HttpRequest | None = context.get("request")
    next_url = get_next_redirect_url(request) if request is not None else None
    if not next_url:
        return None
    try:
        match = resolve(urlparse(next_url).path)
    except Resolver404:
        return None
    if match.view_name != "course_interest:deferred_express_interest":
        return None
    course = Course.objects.filter(slug=match.kwargs["course_slug"]).first()
    if course is None:
        return None
    # Not a view lookup: an unavailable course means "no lead line", not a 404 page.
    try:
        raise_404_if_unavailable(request.user, course)
    except Http404:
        return None
    return course
```
