**1b. Reference query** in `freedom_ls/learner_interface/utils.py`:

```python
@dataclass(frozen=True)
class ReferencePartTopics:
    """A course's reference part with the topics a learner can read in it."""

    course_part: CoursePart
    topics: list[Topic]


def course_reference_topics(
    course: Course, *, part_slug: str | None = None
) -> list[ReferencePartTopics]:
    """The course's reference parts in order, each with its topics in order.

    Reference parts hold topics only, but anything else placed in one is dropped
    rather than served, and a part left with no topics is dropped with it.
    Children are prefetched per content type, so the query count does not grow
    with the number of parts or topics.
    """
    rows = course.reference_parts.select_related("course_part").prefetch_related(
        Prefetch(
            "course_part__items",
            queryset=ContentCollectionItem.objects.prefetch_related("child"),
        )
    )
    if part_slug is not None:
        rows = rows.filter(course_part__slug=part_slug)
    parts = []
    for row in rows:
        topics = [
            item.child
            for item in row.course_part.items.all()
            if isinstance(item.child, Topic)
        ]
        if topics:
            parts.append(ReferencePartTopics(course_part=row.course_part, topics=topics))
    return parts
```

Ordering comes from the models' `Meta.ordering` (`CourseReferencePart.order`,
`ContentCollectionItem.order`). Do not call `CoursePart.children()` or `collection_items()`.

Tests in `tests/test_utils.py`: parts in `CourseReferencePart.order` and topics in
`ContentCollectionItem.order`; a non-topic child (a `Form`) is dropped; a part with only a form, or
no items, is dropped; a part attached only to another course is never returned; `part_slug`
narrows to that part; the query count is the same bound with 1 part × 1 topic and 3 parts × 3
topics (`django_assert_max_num_queries`, constant named at the top of the test module section).

```python
def _course_index_without_current_row(
    request: HttpRequest, course: Course, course_progress: CourseProgress | None
) -> list[dict]:
    """The outline for a page that is not one of the course's items.

    The caller has confirmed content access, so no row is rendered locked for
    want of it.
    """
    return get_course_index(
        user=request.user,
        course=course,
        can_access_content=True,
        course_progress=course_progress,
        course_deadlines=_deadlines_or_none(course_progress),
        deadline_policy=_deadline_policy_or_none(course_progress),
    )


def _reference_access_redirect(
    request: HttpRequest, course: Course
) -> HttpResponseRedirect | None:
    """Where to send a learner who may not read this course's reference pages, or None.

    Reference pages follow the same gate as a closed item: whoever cannot see
    the course's content goes to its detail page. There are no read-only
    exceptions, because reading reference material records nothing.
    """
    decision = _course_access_decision(request.user, course)
    if decision.can_access_content:
        return None
    return redirect("learner_interface:course_detail", course_slug=course.slug)


def _reference_sidebar_context(
    request: HttpRequest, course: Course
) -> dict[str, object]:
    """The outline panel with no current row, read without writing any progress.

    course_progress_for only reads, so a learner who has never opened the
    course still has no course progress record after reading its reference material.
    """
    course_progress = course_progress_for(cast("User", request.user), course)
    return {
        "course_progress": course_progress,
        "course_index": _course_index_without_current_row(
            request, course, course_progress
        ),
        "on_reference_page": True,
        **_time_zone_context(request),
    }


@login_required
def course_reference(request: HttpRequest, course_slug: str) -> HttpResponse:
    course = get_object_or_404(Course, slug=course_slug)
    no_access = _reference_access_redirect(request, course)
    if no_access is not None:
        return no_access
    return render(
        request,
        "learner_interface/course_reference.html",
        {
            "course": course,
            "reference_part_topics": course_reference_topics(course),
            **_reference_sidebar_context(request, course),
        },
    )
```

**2b. View.** The part is found through this course's `CourseReferencePart` rows by passing
`part_slug` to `course_reference_topics`, so a part slug valid only for another course, a part with
no topics, a topic slug outside the part and a non-topic child are all 404s:

```python
@login_required
def course_reference_topic(
    request: HttpRequest, course_slug: str, part_slug: str, topic_slug: str
) -> HttpResponse:
    course = get_object_or_404(Course, slug=course_slug)
    no_access = _reference_access_redirect(request, course)
    if no_access is not None:
        return no_access
    parts = course_reference_topics(course, part_slug=part_slug)
    if not parts:
        raise Http404("No such reference part in this course.")
    part = parts[0]
    position = next(
        (n for n, topic in enumerate(part.topics) if topic.slug == topic_slug), None
    )
    if position is None:
        raise Http404("No such topic in this reference part.")
    topic = part.topics[position]
    # Previous and Next stay inside the part: reference parts are read
    # separately, not as one sequence.
    previous_topic = part.topics[position - 1] if position > 0 else None
    next_topic = part.topics[position + 1] if position + 1 < len(part.topics) else None
    return render(
        request,
        "learner_interface/course_reference_topic.html",
        {
            "course": course,
            "topic": topic,
            "current_part": part.course_part,
            "previous_topic": previous_topic,
            "next_topic": next_topic,
            "previous_url": (
                _reference_topic_url(course, part.course_part, previous_topic)
                if previous_topic
                else None
            ),
            "next_url": (
                _reference_topic_url(course, part.course_part, next_topic)
                if next_topic
                else None
            ),
            **_reference_sidebar_context(request, course),
        },
    )


def _reference_topic_url(course: Course, course_part: CoursePart, topic: Topic) -> str:
    """The reference topic page for one topic of one of the course's reference parts."""
    return reverse(
        "learner_interface:course_reference_topic",
        kwargs={
            "course_slug": course.slug,
            "part_slug": course_part.slug,
            "topic_slug": topic.slug,
        },
    )
```

> **Assumed default:** finding the topic and its neighbours stays inline in the view. Why: it is
> short and the view tests cover every branch. To change: slice 2, 2b.
