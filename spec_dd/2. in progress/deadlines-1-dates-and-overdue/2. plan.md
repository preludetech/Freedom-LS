## Slice 2: site time zone

**Behaviour.** A superuser opens the admin, adds a "Site time zone" row for the site with
`Africa/Johannesburg`, and saves; a second row for the same site is refused; `Mars/Olympus` is

```python
def validate_time_zone(value: str) -> None:
    """Only a name the zoneinfo database knows can convert a due date to an instant."""
    try:
        ZoneInfo(value)
    except (ZoneInfoNotFoundError, ValueError) as exc:
        raise ValidationError("Not an IANA time zone name, e.g. Africa/Johannesburg.") from exc


class SiteTimeZoneConfig(SiteAwareModel, TimestampedModel):
    """The zone a site's authored dates are read in. No row means settings.TIME_ZONE."""

    time_zone = models.CharField(max_length=64, validators=[validate_time_zone])
    show_time_zone_to_learners = models.BooleanField(
        default=False,
        help_text="Label every learner-facing date with its time zone.",
    )

    class Meta:
        constraints = [
            models.UniqueConstraint(fields=["site"], name="unique_time_zone_config_per_site"),
        ]
        verbose_name = "site time zone"

    def __str__(self) -> str:
        return f"{self.site.domain}: {self.time_zone}"
```

## Slice 3: content defaults authored, validated and loaded

**Behaviour.** A topic or form file carries `due_date`, `due_time` and

**Files.**

- `freedom_ls/content_base/schema.py`:

```python
_DUE_TIME = re.compile(r"\d{2}:\d{2}")


class DueDateFields(BaseModel):
    """The deadline a child file carries: a fixed date, or days after the learner's registration
    date, either with an optional time of day. Pure pydantic so the offline validator shares it."""

    due_date: date | None = Field(None, description="Fixed due date (YAML date)")
    due_time: time | None = Field(None, description='Time of day, 24-hour "HH:MM"')
    due_days_after_registration: int | None = Field(
        None, ge=0, strict=True, description="Days after the registration date"
    )

    @field_validator("due_date", mode="before")
    @classmethod
    def _due_date_is_a_date(cls, value: object) -> object:
        # PyYAML parses `2026-03-14 00:00:00` as a datetime and pydantic's lax date would
        # accept it; a due date never carries a time of its own, that is what due_time is for.
        if value is not None and type(value) is not date:
            raise ValueError("due_date must be a date such as 2026-03-14, without a time")
        return value

    @field_validator("due_time", mode="before")
    @classmethod
    def _due_time_is_quoted(cls, value: object) -> object:
        # An unquoted 17:00 is a YAML sexagesimal integer (1020), which pydantic would read
        # as seconds after midnight. Only the quoted string says what the author meant.
        if value is not None and not (isinstance(value, str) and _DUE_TIME.fullmatch(value)):
            raise ValueError('due_time must be a quoted 24-hour time such as "17:00"')
        return value

    @model_validator(mode="after")
    def _one_kind_of_due_date(self) -> "DueDateFields":
        if self.due_date is not None and self.due_days_after_registration is not None:
            raise ValueError(
                f"{self.file_path}: set due_date or due_days_after_registration, not both"
            )
        if self.due_time is not None and self.due_date is None and self.due_days_after_registration is None:
            raise ValueError(f"{self.file_path}: due_time needs due_date or due_days_after_registration")
        return self


DUE_DATE_FIELD_NAMES = frozenset(DueDateFields.model_fields)
```

  `file_path` is on `BaseBaseContentModel`; the validator runs on the combined class, so the
  attribute exists. A malformed quoted time (`"25:99"`) is refused by pydantic's own `time`
  parsing.

```python
class DeadlineSource(StrEnum):
    CONTENT = "content"
    COHORT_REGISTRATION = "cohort_registration"
    LEARNER = "learner"


@dataclass(frozen=True)
class DeadlineValue:
    kind: DeadlineKind
    due_date: date | None = None
    due_time: time | None = None
    due_days_after_registration: int | None = None

    @classmethod
    def from_columns(
        cls, due_date: date | None, due_time: time | None, due_days_after_registration: int | None
    ) -> DeadlineValue | None:
        """The value three nullable columns express, or None when they hold no deadline."""
        if due_date is not None:
            return cls(DeadlineKind.FIXED, due_date=due_date, due_time=due_time)
        if due_days_after_registration is not None:
            return cls(
                DeadlineKind.RELATIVE,
                due_time=due_time,
                due_days_after_registration=due_days_after_registration,
            )
        return None

    def canonical(self) -> str:
        """One string per distinct value, so a stored copy can be compared with a later read."""
        if self.kind == DeadlineKind.NONE:
            return "none"
        suffix = f"T{self.due_time:%H:%M}" if self.due_time is not None else ""
        if self.kind == DeadlineKind.FIXED:
            if self.due_date is None:
                raise ValueError("A fixed deadline carries a date.")
            return f"fixed:{self.due_date.isoformat()}{suffix}"
        return f"relative:{self.due_days_after_registration}{suffix}"


class DeadlineLayer(NamedTuple):
    source: DeadlineSource
    value: DeadlineValue | None
    replaced_value: str | None


@dataclass(frozen=True)
class ResolvedDeadline:
    due_at: datetime | None
    source: DeadlineSource | None
    default_changed: bool


NO_DEADLINE = ResolvedDeadline(due_at=None, source=None, default_changed=False)


def due_instant(value: DeadlineValue, registration_date: date, tz: ZoneInfo) -> datetime | None:
    """The instant a value falls due for a learner registered on registration_date.

    Relative offsets are added to the local date, not to an instant, so a DST change between
    registration and the due day still lands on the local end of day. A value without a time
    is due at the last microsecond of its day."""
    if value.kind == DeadlineKind.NONE:
        return None
    if value.kind == DeadlineKind.FIXED:
        if value.due_date is None:
            raise ValueError("A fixed deadline carries a date.")
        day = value.due_date
    else:
        if value.due_days_after_registration is None:
            raise ValueError("A relative deadline carries a number of days.")
        day = registration_date + timedelta(days=value.due_days_after_registration)
    at = value.due_time if value.due_time is not None else time.max
    return datetime.combine(day, at, tzinfo=tz)


def resolve_deadline(
    layers: Sequence[DeadlineLayer], registration_date: date, tz: ZoneInfo
) -> ResolvedDeadline:
    """Most specific layer first; the first layer holding a value wins outright.

    default_changed compares the winning layer's stored copy of what it replaced with what
    sits beneath it now, so an admin can see that a content default moved under an override."""
    for position, layer in enumerate(layers):
        if layer.value is None:
            continue
        beneath = next((lower.value for lower in layers[position + 1 :] if lower.value is not None), None)
        beneath_canonical = beneath.canonical() if beneath is not None else ""
        return ResolvedDeadline(
            due_at=due_instant(layer.value, registration_date, tz),
            source=layer.source,
            default_changed=layer.replaced_value is not None
            and layer.replaced_value != beneath_canonical,
        )
    return NO_DEADLINE


def is_overdue(resolved: ResolvedDeadline, completed: bool, now: datetime) -> bool:
    return resolved.due_at is not None and not completed and now > resolved.due_at


def registration_date_of(course_progress: CourseProgress, tz: ZoneInfo) -> date:
    """The date part of the record's creation, in the site's zone."""
    return course_progress.created_at.astimezone(tz).date()


@dataclass(frozen=True)
class CourseDeadlines:
    items: dict[UUID, ResolvedDeadline]
    time_limit: ResolvedDeadline
```

```python
LayerRows = Mapping[tuple[UUID, UUID | None], "CohortDeadlineOverride | LearnerDeadlineOverride"]


@dataclass(frozen=True)
class _CourseDefaults:
    """What the content says for one course: a value per viewable row, and the time limit."""

    items: dict[UUID, DeadlineValue | None]
    time_limit: DeadlineValue | None


def _defaults_by_course(course_ids: set[UUID]) -> dict[UUID, _CourseDefaults]:
    """Three queries whatever the number of courses or rows: the course-level rows, the
    part-level rows beneath them, and the courses' time limits. Rows whose child is a part are
    containers, not items, so they carry no value."""
    course_type = ContentType.objects.get_for_model(Course)
    part_type = ContentType.objects.get_for_model(CoursePart)
    top_rows = list(ContentCollectionItem.objects.filter(collection_type=course_type, collection_id__in=course_ids))
    course_of_part = {row.child_id: row.collection_id for row in top_rows if row.child_type_id == part_type.pk}
    part_rows = list(ContentCollectionItem.objects.filter(collection_type=part_type, collection_id__in=course_of_part))
    time_limit_days = dict(Course.objects.filter(pk__in=course_ids).values_list("pk", "time_limit_days"))

    defaults = {
        course_id: _CourseDefaults(items={}, time_limit=_time_limit_value(time_limit_days.get(course_id)))
        for course_id in course_ids
    }
    for row in top_rows + part_rows:
        if row.child_type_id == part_type.pk:
            continue
        course_id = row.collection_id if row.collection_type_id == course_type.pk else course_of_part[row.collection_id]
        defaults[course_id].items[row.pk] = DeadlineValue.from_columns(
            row.due_date, row.due_time, row.due_days_after_registration
        )
    return defaults


def _time_limit_value(days: int | None) -> DeadlineValue | None:
    return DeadlineValue(DeadlineKind.RELATIVE, due_days_after_registration=days) if days is not None else None


def _layer(source: DeadlineSource, row: CohortDeadlineOverride | LearnerDeadlineOverride | None) -> DeadlineLayer:
    return DeadlineLayer(source, row.value() if row else None, row.replaced_value if row else None)


def _layers(
    record: CourseProgress, item_id: UUID | None, content: DeadlineValue | None,
    cohort_rows: LayerRows, learner_rows: LayerRows,
) -> list[DeadlineLayer]:
    cohort_row = cohort_rows.get((record.cohort_registration_id, item_id)) if record.cohort_registration_id else None
    return [
        _layer(DeadlineSource.LEARNER, learner_rows.get((record.pk, item_id))),
        _layer(DeadlineSource.COHORT_REGISTRATION, cohort_row),
        DeadlineLayer(DeadlineSource.CONTENT, content, None),
    ]


def course_deadlines(course_progress: CourseProgress) -> CourseDeadlines:
    return course_deadlines_for([course_progress])[course_progress.pk]


def course_deadlines_for(
    course_progress_records: Iterable[CourseProgress],
) -> dict[UUID, CourseDeadlines]:
    """Every record's resolved deadlines, keyed by record id.

    A fixed number of queries whatever the number of records or collection items. The site
    zone is read once; every record a request hands in belongs to the request's site."""
    records = list(course_progress_records)
    if not records:
        return {}
    tz = site_time_zone(records[0].site)
    defaults = _defaults_by_course({record.course_id for record in records})
    cohort_rows: LayerRows = {}
    learner_rows: LayerRows = {}

    result: dict[UUID, CourseDeadlines] = {}
    for record in records:
        registered_on = registration_date_of(record, tz)
        course_defaults = defaults[record.course_id]
        items = {
            item_id: resolve_deadline(_layers(record, item_id, content, cohort_rows, learner_rows), registered_on, tz)
            for item_id, content in course_defaults.items.items()
        }
        time_limit = resolve_deadline(
            _layers(record, None, course_defaults.time_limit, cohort_rows, learner_rows), registered_on, tz
        )
        result[record.pk] = CourseDeadlines(items=items, time_limit=time_limit)
    return result
```

```python
@dataclass(frozen=True)
class CardDeadlineLine:
    """What the card's one deadline line says, in the order the template checks it."""

    time_limit_end: datetime | None
    time_limit_overdue: bool
    overdue_count: int
    next_due_at: datetime | None


def _card_deadline_line(flat: list[dict], course_deadlines: CourseDeadlines | None, now: datetime) -> CardDeadlineLine:
    # Only leaf rows carry due_at; a completed leaf neither counts as overdue nor sets the
    # next due date. An in-progress course is never complete, so the time limit's overdue
    # check takes completed=False.
    leaves = [row for row in flat if "due_at" in row]
    open_due_dates = [row["due_at"] for row in leaves if row["due_at"] and row["status"] != COMPLETE]
    time_limit = course_deadlines.time_limit if course_deadlines else NO_DEADLINE
    upcoming = [due_at for due_at in open_due_dates if due_at > now]
    if time_limit.due_at is not None and time_limit.due_at > now:
        upcoming.append(time_limit.due_at)
```

## Slice 6a: cohort deadline override

**Behaviour.** On a `CohortCourseRegistration` change page an admin adds a "Deadline override"
row: picks a collection item (or "Course time limit"), a kind, a date or a number of days, an
optional time, a reason, and saves; the row records who set it and the canonical value it
replaced; it shows the content default and the effective value read-only. After the content
default under it changes and is re-synced, the row shows "Default has changed since this override
was set (was X, now Y)". A learner in the cohort sees the override beat the content default on the
TOC, course detail and the dashboard card; a learner on an individual registration for the same
course does not. A `none` override removes the date. `clean()` refuses each inconsistent shape, a
collection item of another course, and an id that is not a UUID in the picker's URL does not 500.
Another site's rows never apply.

```python
def _validate_deadline_override(
    kind: str, due_date: date | None, due_time: time | None,
    due_days_after_registration: int | None, collection_item: ContentCollectionItem | None,
    course: Course,
) -> None:
    """The shape rules every override row obeys, raised as one ValidationError."""
    errors: dict[str, str] = {}
    if kind == DeadlineKind.FIXED and due_date is None:
        errors["due_date"] = "A fixed deadline needs a date."
    if kind == DeadlineKind.RELATIVE and due_days_after_registration is None:
        errors["due_days_after_registration"] = "A relative deadline needs a number of days."
    if kind == DeadlineKind.NONE and (due_date or due_time or due_days_after_registration is not None):
        errors["kind"] = "No deadline takes no date, time or days."
    if collection_item is None:
        if kind == DeadlineKind.FIXED:
            errors["kind"] = "The course time limit is counted in days after registration."
        if due_time is not None:
            errors["due_time"] = "The course time limit ends at the end of its day."
    elif collection_item not in course_viewable_collection_items(course):
        errors["collection_item"] = "Pick an item of this course."
    if errors:
        raise ValidationError(errors)

    collection_item = models.ForeignKey(ContentCollectionItem, null=True, blank=True, on_delete=models.CASCADE, related_name="+")
    kind = models.CharField(max_length=8, choices=DeadlineKind.choices)
    due_date = models.DateField(null=True, blank=True, help_text="Read in the site's time zone.")
    due_time = models.TimeField(null=True, blank=True)
    due_days_after_registration = models.PositiveIntegerField(null=True, blank=True, verbose_name="days after registration")
    replaced_value = models.CharField(max_length=64, blank=True)
    set_by = models.ForeignKey(settings.AUTH_USER_MODEL, null=True, blank=True, on_delete=models.SET_NULL, related_name="+")
    reason = models.TextField(blank=True)

    class Meta:
        constraints = [
            models.UniqueConstraint(fields=["cohort_registration", "collection_item"], condition=Q(collection_item__isnull=False), name="unique_cohort_override_per_item"),
            models.UniqueConstraint(fields=["cohort_registration"], condition=Q(collection_item__isnull=True), name="unique_cohort_time_limit_override"),
        ]

        # The parent is unset while the admin add page is still being validated.
        return self.cohort_registration.course if self.cohort_registration_id else None

        if self.course is None:
            return
```

```python
class DeadlineOverrideInline(TabularInline):
    """Shared shape of the two override inlines; the subclass names the model and how to reach
    the parent's course."""

    extra = 0
    formset = DeadlineOverrideFormSet
    fields = ["collection_item", "kind", "due_date", "due_time", "due_days_after_registration",
              "reason", "content_default", "effective_value", "default_changed_note", "set_by"]
    readonly_fields = ["content_default", "effective_value", "default_changed_note", "set_by"]
    verbose_name = "deadline override"

    def parent_course(self, request: HttpRequest) -> Course | None:
        """The course of the object being edited, or None on the add page or for a malformed id.
        Read through the site-aware manager, so another site's parent is never reached."""
        object_id = request.resolver_match.kwargs.get("object_id") if request.resolver_match else None
        if object_id is None:
            return None
        try:
            parent = self.parent_model.objects.select_related("course").filter(pk=unquote(object_id)).first()
        except (ValidationError, ValueError):
            return None
        return parent.course if parent is not None else None

    def get_formset(self, request, obj=None, **kwargs):
        formset = super().get_formset(request, obj, **kwargs)
        formset.set_by = request.user
        return formset

    def formfield_for_foreignkey(self, db_field, request, **kwargs):
        # The picker offers only the parent course's own rows. On the add page there is no
        # course yet, so it offers nothing; clean() refuses a row from another course regardless.
        if db_field.name == "collection_item":
            course = self.parent_course(request)
            kwargs["form_class"] = CollectionItemChoiceField
            kwargs["empty_label"] = "Course time limit"
            kwargs["queryset"] = (
                ContentCollectionItem.objects.filter(pk__in=[i.pk for i in course_viewable_collection_items(course)])
                if course is not None else ContentCollectionItem.objects.none()
            )
        return super().formfield_for_foreignkey(db_field, request, **kwargs)

    @admin.display(description="Content default")
    def content_default(self, obj) -> str: ...   # "14 days after registration" / "Fixed 14 Mar 2026, 17:00 SAST" / "No deadline" / "None"

    @admin.display(description="Effective")
    def effective_value(self, obj) -> str: ...   # resolve_deadline over [obj's own layer, *layers_beneath(obj)], rendered with site_zone_text

    @admin.display(description="Default changed")
    def default_changed_note(self, obj) -> str:
        if not obj.pk:
            return ""
        beneath = canonical_beneath(obj)
        if obj.replaced_value == beneath:
            return ""
        return f"Default has changed since this override was set (was {obj.replaced_value or 'nothing'}, now {beneath or 'nothing'})"
```

## Slice 7: deadline override carry-over on re-sync

**Behaviour.** A topic with a collection item in part A carries a cohort override and a learner
override. The author moves the file to part B and re-syncs. The load creates a new
`ContentCollectionItem` under B, the old one under A stays (today's sync never prunes), and both
deadline override rows now point at the new row; the learner sees the same deadline on the moved
topic. A file with two collection items on purpose, a child with two stale rows, another site's
row for the same child, or a load that fails after the move leaves every row where it was.

**Files.**

- New `freedom_ls/content_engine/signals.py`:

```python

# transaction, with `site`, `created` (the ContentCollectionItem rows this load made) and
# `written` (every row it touched, created ones included). A receiver that raises rolls the
# load back.
collection_items_synced = Signal()
```

- `content_save.py` child pass: collect `(row, was_created)` from `update_or_create` into

  `collection_items_synced.send(sender=ContentCollectionItem, site=site, created=created, written=written)`.

```python
@receiver(collection_items_synced)
def carry_over_deadline_overrides(
    sender: type[ContentCollectionItem], site: Site,
    created: list[ContentCollectionItem], written: list[ContentCollectionItem], **kwargs: object,
) -> None:
    """Move deadline override rows from a child's stale collection item to the one a load just
    made.

    A sync never removes the row a moved child left behind, so the deadline overrides pointing
    at it would otherwise stop applying. Only the unambiguous case moves: exactly one other row
    for the same child in the same course that this load did not write. A file with two
    collection items on purpose has both rows written, and a child with two stale rows is left
    for a person."""
    written_ids = {row.pk for row in written}
    for new_row in created:
        course = _course_of_collection_item(new_row, site)
        if course is None:
            continue
        stale = [
            row for row in course_viewable_collection_items(course)
            if row.pk != new_row.pk and row.pk not in written_ids and row.site_id == site.pk
            and (row.child_type_id, row.child_id) == (new_row.child_type_id, new_row.child_id)
        ]
        if len(stale) != 1:
            continue
        for model in (CohortDeadlineOverride, LearnerDeadlineOverride):
            model._base_manager.filter(site=site, collection_item=stale[0]).update(collection_item=new_row)


def _course_of_collection_item(row: ContentCollectionItem, site: Site) -> Course | None:
    """The course a row sits in: its collection when that is a course, else the course that
    places its part. Read through _base_manager with the load's site, because a load runs with
    no request; a fresh instance each time, so the memoised item tree is this load's."""
```
