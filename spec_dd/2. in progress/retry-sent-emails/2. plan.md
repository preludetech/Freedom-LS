> **Decision:** this stays one slice. The existing backend and course-application tests expect
> the inline failure report, so a first slice without it would leave the suite failing.

the row id. The task claims the row, sends it and marks it `sent` with the payload removed. Any

   ```python
   def _send_email_task(email_id: str) -> None:

   def enqueue_send(email_id: UUID, run_after: datetime | None = None) -> TaskResult:
       """Enqueue one attempt at sending a row, optionally not before run_after.

       The task carries the row id and nothing else, so no copy of the message
       body is written to the task table.
       """
       task = _send_email_task
       if run_after is not None:
           task = task.using(run_after=run_after)
       return task.enqueue(str(email_id))


   def send_outbound_email(email_id: str) -> None:
       now = timezone.now()
       # One UPDATE is the whole guard: a second worker, a redelivered task, a
       # sweep and a resend can all race for the row and exactly one wins. The
       # due-time filter stops a stray task sending before the row is due.
       claimed = OutboundEmail.objects.filter(
           pk=email_id,
           status__in=(Status.PENDING, Status.FAILED),
           next_attempt_at__lte=now,
       ).update(
           status=Status.SENDING,
           attempt_count=F("attempt_count") + 1,
           last_attempt_at=now,
       )
       if not claimed:
           return
       email = OutboundEmail.objects.get(pk=email_id)
       send_serialised_email(email.payload)
       # Nothing runs between the send and this write: a crash here re-sends
       # the message, so the window is kept as small as one statement.
       sent_at = timezone.now()
       email.status = Status.SENT
       email.sent_at = sent_at
       email.payload = {}
       email.scrubbed_at = sent_at
       email.last_error = ""
       email.save(update_fields=[...the five fields...])
   ```

### Slice 2: a permanent failure stops at one attempt

A send that fails with an error not worth retrying marks the row `permanent_failure`, records
`last_error`, raises one constant-text Sentry message, and the task finishes normally. Transient
and unclassified errors still propagate, and now leave their cause in `last_error`.

**Builds on:** slices 1
**Touches:** freedom_ls/mail/errors.py, freedom_ls/mail/tasks.py, freedom_ls/mail/tests/conftest.py, freedom_ls/mail/tests/test_errors.py, freedom_ls/mail/tests/test_tasks.py

1. `freedom_ls/mail/errors.py`. Top-level imports are stdlib, `sentry_sdk` and
   `django.core.mail.BadHeaderError` only. `OutboundEmail` is imported under `TYPE_CHECKING`.

   ```python
   """Deciding whether a failed send is worth another attempt.

   One blind spot: smtplib.sendmail raises only when every recipient is refused,
   so a message some recipients refused is recorded as sent and not retried. A
   retry would send a second copy to the recipients who accepted the first.
   """

   class SendErrorKind(StrEnum):
       TRANSIENT = "transient"
       PERMANENT = "permanent"


   class UpstreamSendCountZeroError(Exception):
       """The upstream backend raised nothing and reported no message sent."""


   def classify_send_error(exc: BaseException) -> SendErrorKind | None:
       if _is_permanent(exc):
           return SendErrorKind.PERMANENT
       if _is_transient(exc):
           return SendErrorKind.TRANSIENT
       return None


   def _is_permanent(exc: BaseException) -> bool:
       # Bad credentials are permanent whatever the code: retrying them across a
       # backlog gets the sending account locked.
       if isinstance(exc, (SMTPAuthenticationError, SMTPNotSupportedError,
                           BadHeaderError, ssl.SSLCertVerificationError,
                           UnicodeEncodeError)):
           return True
       if isinstance(exc, SMTPRecipientsRefused):
           return any(500 <= code <= 599 for code, _ in exc.recipients.values())
       smtp_code = getattr(exc, "smtp_code", None)
       return isinstance(smtp_code, int) and 500 <= smtp_code <= 599


   def _is_transient(exc: BaseException) -> bool:
       # Every SMTPException is an OSError, and so is each socket and TLS error
       # worth retrying. Permanent cases are ruled out before this runs.
       return isinstance(exc, (OSError, UpstreamSendCountZeroError))


   def describe_error(exc: BaseException) -> str:
       """'<exception class path>: <message>', the text stored in last_error."""
       cls = type(exc)
       return f"{cls.__module__}.{cls.__qualname__}: {exc}"


   def report_outbound_email(email: OutboundEmail, message: str) -> None:
       """Send Sentry a constant message about one row.

       Only the row id, attempt count and status go on the scope. Exception text
       from smtplib can carry recipient addresses, so it is never attached, and a
       constant message groups every such row into one issue.
       """
       with sentry_sdk.new_scope() as scope:
           scope.set_extra("outbound_email_id", str(email.pk))
           scope.set_extra("attempt_count", email.attempt_count)
           scope.set_extra("status", email.status)
           sentry_sdk.capture_message(message)
   ```

### Slice 3: a transient failure is retried on a backoff ladder, then dead-lettered

A send that fails with a transient error marks the row `failed`, sets `next_attempt_at` from the
ladder and enqueues its own next attempt when the task backend can defer. At
`EMAIL_MAX_ATTEMPTS` the row goes to `dead_letter` with one Sentry message. An upstream backend
that reports 0 messages sent counts as a transient failure.

**Builds on:** slices 1, 2
**Touches:** freedom_ls/mail/config.py, freedom_ls/mail/retry.py, freedom_ls/mail/tasks.py, freedom_ls/mail/tests/conftest.py, freedom_ls/mail/tests/test_retry.py, freedom_ls/mail/tests/test_tasks.py, freedom_ls/mail/tests/test_backends.py, freedom_ls/mail/tests/test_config.py

   `EMAIL_RETRY_DELAYS: list[int]` (`[60, 300, 900, 1800, 3600]`) and
   `EMAIL_RETRY_JITTER_RATIO: float` (0.2), each with a comment saying what it bounds.
2. `freedom_ls/mail/retry.py`:

   ```python
   def next_attempt_delay(attempt_count: int) -> timedelta:
       """How long to wait after the given attempt failed."""
       delays = config.EMAIL_RETRY_DELAYS
       base = delays[min(attempt_count - 1, len(delays) - 1)]
       # Jitter spreads a backlog that all failed in the same second, so the
       # retries do not arrive at the mail host as one burst.
       jitter = random.uniform(0, base * config.EMAIL_RETRY_JITTER_RATIO)  # noqa: S311
       return timedelta(seconds=base + jitter)


   def schedule_retry(email: OutboundEmail, exc: BaseException) -> None:
       # Imported here because tasks imports this module.
       from freedom_ls.mail.tasks import enqueue_send

       email.last_error = describe_error(exc)
       if email.attempt_count >= config.EMAIL_MAX_ATTEMPTS:
           email.status = OutboundEmail.Status.DEAD_LETTER
           email.save(update_fields=["status", "last_error"])
           report_outbound_email(email, DEAD_LETTER_MESSAGE)
           return
       email.status = OutboundEmail.Status.FAILED
       email.next_attempt_at = timezone.now() + next_attempt_delay(email.attempt_count)
       email.save(update_fields=["status", "next_attempt_at", "last_error"])
       # A backend that cannot defer would run the next attempt at once, and an
       # inline one would walk the whole ladder inside the request. Housekeeping
       # drives the row instead.
       if default_task_backend.supports_defer:
           enqueue_send(email.pk, run_after=email.next_attempt_at)
   ```

   ```python
   def requeue_due_outbound_email() -> int:
       """Enqueue a send for each row that is overdue. Returns how many."""
       now = timezone.now()
       cutoff = now - timedelta(seconds=config.HOUSEKEEPING_DUE_EMAIL_GRACE_SECONDS)
       overdue = OutboundEmail.objects.filter(
           status__in=(Status.PENDING, Status.FAILED), next_attempt_at__lt=cutoff
       )
       due_ids = list(
           overdue.order_by("next_attempt_at")
           .values_list("pk", flat=True)[: config.HOUSEKEEPING_EMAIL_REQUEUE_BATCH_SIZE]
       )
       requeued = 0
       for email_id in due_ids:
           # One transaction per row: an enqueue that fails rolls the due time
           # back, so the row is still overdue on the next run.
           with transaction.atomic():
               # Moving the due time to now takes the row out of the next run's
               # match, so a run made while the worker is down does not stack a
               # second task behind the first.
               if overdue.filter(pk=email_id).update(next_attempt_at=now):
                   enqueue_send(email_id)
                   requeued += 1
       return requeued
   ```

   ```python
   def recover_stuck_outbound_email() -> list[OutboundEmail]:
       """Re-drive rows a dead worker left in sending. Returns the rows dead-lettered."""
       now = timezone.now()
       cutoff = now - timedelta(seconds=config.HOUSEKEEPING_STUCK_EMAIL_MAX_AGE_SECONDS)
       dead_lettered: list[OutboundEmail] = []
       stuck = OutboundEmail.objects.filter(status=Status.SENDING, last_attempt_at__lt=cutoff)
       for email in stuck:
           out_of_attempts = email.attempt_count >= config.EMAIL_MAX_ATTEMPTS
           new_status = Status.DEAD_LETTER if out_of_attempts else Status.FAILED
           # Conditional on the attempt this run saw, so a row a worker finished
           # or re-claimed since the query is left to that worker.
           moved = OutboundEmail.objects.filter(
               pk=email.pk, status=Status.SENDING, last_attempt_at=email.last_attempt_at
           ).update(
               status=new_status,
               next_attempt_at=now,
               # A row whose task died without recording a cause still says
               # something in the admin.
               last_error=email.last_error or STUCK_LAST_ERROR,
           )
           if not moved:
               continue
           email.status = new_status
           report_outbound_email(email, STUCK_MESSAGE)
           if out_of_attempts:
               dead_lettered.append(email)
           else:
               enqueue_send(email.pk)
       return dead_lettered
   ```

### Slice 8: a superuser can resend failed rows from the changelist

1. `freedom_ls/mail/admin.py`. One function holds the rule for whether a row can be resent, so
   the bulk action, the detail button and the confirmation page cannot disagree.

   ```python
   RESENDABLE_STATUSES = (Status.PERMANENT_FAILURE, Status.DEAD_LETTER)


   def resend_blocker(email: OutboundEmail) -> str | None:
       """Why this row cannot be resent, or None when it can."""
       if email.status == Status.SENT:
           return "already sent"
       if email.status not in RESENDABLE_STATUSES:
           return "in flight"
       if email.scrubbed_at is not None:
           return "body removed"
       return None
   ```

   On `OutboundEmailAdmin`:

   ```python
   def has_resend_permission(
       self, request: HttpRequest, obj: OutboundEmail | None = None
   ) -> bool:
       return request.user.is_superuser

   @admin.action(description="Resend selected emails", permissions=["resend"])
   def resend_selected(self, request: HttpRequest, queryset: QuerySet[OutboundEmail]) -> None:
       self._resend(request, queryset)

   def _resend(self, request: HttpRequest, emails: Iterable[OutboundEmail]) -> None:
       """Re-queue each resendable row and tell the admin what happened to the rest."""
       queued = 0
       skipped: Counter[str] = Counter()
       now = timezone.now()
       for email in emails:
           blocker = resend_blocker(email)
           if blocker is not None:
               skipped[blocker] += 1
               continue
           with transaction.atomic():
               # Conditional, so two admins resending the same row queue it once.
               # attempt_count returns to 0: a resend is a new ladder, not one
               # more step on the ladder that already ran out.
               moved = OutboundEmail.objects.filter(
                   pk=email.pk, status__in=RESENDABLE_STATUSES, scrubbed_at__isnull=True
               ).update(status=Status.PENDING, attempt_count=0, next_attempt_at=now)
               if not moved:
                   skipped["in flight"] += 1
                   continue
               self.log_change(request, email, "Resent.")
               transaction.on_commit(partial(enqueue_send, email.pk), robust=True)
           queued += 1
       ...self.message_user at messages.SUCCESS with the number queued, when any...
       ...self.message_user at messages.WARNING with the number skipped and each
          non-zero reason with its count, when any...
   ```
