3. **Player factory.** In `alpine-components.js`, inside the `alpine:init` callback (it needs the
   closure's `owner` and `audioElement`):

   ```js
   // The player half of an audio widget: its state, and taking or giving up
   // the page's one audio element. Each widget gets a fresh copy, so the owner
   // hand-off exists once and one file plays at a time across every widget.
   function audioPlayer() {
       return {
           _id: nextAudioFileId++,
           idle: true,
           loading: false,
           playing: false,
           failed: false,

           toggle() {
               if (this._sounding()) this._stop();
               else this._start(this.$el.dataset.src);
           },
           errorText() {
               return this.failed ? "Audio could not be played. Try again." : "";
           },
           // Takes the element from whoever owns it and plays from the start.
           // Never stops, so a second call while playing restarts the clip.
           _start(src) {
               // the current toggle() body from `if (owner) {` through
               // `audio.play().catch(...)`, with its comments, reading `src`
               // from the argument instead of this.$el
           },
           _stop() {
               const audio = audioElement();
               audio.pause();
               audio.currentTime = 0;
               this._setState("idle");
           },
           _sounding() {
               return owner?._id === this._id && !this.idle;
           },
           destroy() { /* unchanged */ },
           _fail() { /* unchanged */ },
           _setState(state) { /* unchanged */ },
       };
   }

   Alpine.data("audioFile", audioPlayer);
   ```

   `audio.html` keeps its inline `failed ? … : ''` binding so its output does not move;
   `errorText` is for widgets written after it.

   ```python
   # The six slot widgets, plus c-slot and the layout wrappers c-grid and
   # c-image-grid that can hold them. md_in_html converts the body of an element
   # marked markdown="1" only when its parent is marked too, so a card inside a
   # grid needs the grid marked. Widget children of a marked element stay raw.
   MARKDOWN_BODY_TAGS: tuple[str, ...] = (
       "c-accordion", "c-admonition", "c-card", "c-flashcard", "c-pull-quote",
       "c-table", "c-slot", "c-grid", "c-image-grid",
   )

   # An opening tag at the start of a line or straight after another tag, so a
   # tag quoted in inline code (`<c-card>`) is left alone.
   _MARKDOWN_BODY_OPENING = re.compile(
       r"(^|>)(\s*)<(" + "|".join(map(re.escape, MARKDOWN_BODY_TAGS)) + r")(?=[\s>/])",
       re.MULTILINE,
   )


   class _MarkMarkdownBodies(Preprocessor):
       """Ask md_in_html to convert the body of every widget whose body is markdown."""

       def run(self, lines: list[str]) -> list[str]:
           text = _MARKDOWN_BODY_OPENING.sub(r'\1\2<\3 markdown="1"', "\n".join(lines))
           return text.split("\n")
   ```
