```python
def prefetch_collection_items(collections: Iterable[Course | CoursePart]) -> None:
    """Fill each collection's collection_items() memo, and its nested parts', in
    one query per nesting level plus one per child content type, instead of a
    query per collection."""
    pending = [c for c in collections if not hasattr(c, "_collection_items_cache")]
    while pending:
        # group pending by content type -> {pk: collection}
        # one ContentCollectionItem query: Q(collection_type=ct, collection_id__in=pks) OR'd per type,
        #   .prefetch_related("child"), ordered by the model's "order"
        # bucket items by (collection_type_id, collection_id); set every pending
        #   collection's _collection_items_cache (empty list when it has none)
        # pending = the CourseParts among the children not yet memoized
```

`indexes.build_course_catalogue`: call `prefetch_collection_items` on the registrations' courses,
deduplicated by `course_id` (one `Course` instance per course, reused for every registration of
it), before the traversal loop.

```python
def active_registrations(
    learner: Learner,
) -> list[LearnerCourseRegistration | CohortCourseRegistration]:
    """The learner's active individual registrations, then the active
    registrations of their active cohorts, read from prefetched relations only."""


def _record_id_by_registration(
    rows: Iterable[tuple[UUID, UUID | None, UUID | None]],
) -> dict[UUID, UUID]:
    """Each registration's course progress record id, from
    (record id, cohort registration id, learner registration id) rows."""
    return {
        cohort_registration_id or learner_registration_id: record_id
        for record_id, cohort_registration_id, learner_registration_id in rows
    }


def _load_course_inputs(
    registrations: Sequence[LearnerCourseRegistration | CohortCourseRegistration],
    learner_ids: list[UUID],
    site_id: int,
) -> tuple[CourseCatalogue, ProgressIndex]:
    # One registration per course, so each course is traversed once.
    one_per_course = list({r.course_id: r for r in registrations}.values())
    catalogue = build_course_catalogue(one_per_course)
    progress = load_progress_index(
        site_id, registrations, learner_ids, catalogue, keyed_by="course_progress"
    )
    return catalogue, progress


def gather_course_details_for_learners(
    learners: Sequence[Learner], site_id: int
) -> dict[UUID, list[LearnerCourseDetail]]:
    """Each learner's course details, one per registration, for a whole page of
    learners in a fixed number of queries.

    Reads the registrations and course progress records LearnerDataTable
    prefetches, so it adds no query per learner.
    """
    now = timezone.now()
    registrations_by_learner = {
        learner.pk: active_registrations(learner) for learner in learners
    }
    every_registration = [
        registration
        for registrations in registrations_by_learner.values()
        for registration in registrations
    ]
    catalogue, progress = _load_course_inputs(
        every_registration, [learner.pk for learner in learners], site_id
    )
    details: dict[UUID, list[LearnerCourseDetail]] = {}
    for learner in learners:
        record_ids = _record_id_by_registration(
            (record.pk, record.cohort_registration_id, record.learner_registration_id)
            for record in learner.course_progress_records.all()
        )
        details[learner.pk] = [
            _build_course_detail(
                registration, record_ids.get(registration.pk), catalogue, progress, now
            )
            for registration in registrations_by_learner[learner.pk]
        ]
    return details
```

- `gather_learner_course_details` keeps its signature and its one record query, and builds its
  inputs through `_record_id_by_registration(<its values_list rows>)` and `_load_course_inputs`.
- `active_registrations` lives in `gather.py` so `reports` never imports `educator_interface`.
  `learner_course_summaries` in `data_tables.py` calls it too, so the two never disagree about which
  courses a learner holds.
- `_build_course_detail` reads `registration.cohort.name` for a cohort registration: the reverse-FK
  prefetch already caches `cohort` on each registration; the query-count test confirms it.
- `build_course_catalogue` keys `course_items` by `course_id`, so the deduplicated catalogue answers
  every registration of a course.

```python
@dataclass(frozen=True)
class RankedCourseSummary:
    """One course in a learner's Courses cell: recomputed completion, whether
    any of its registrations is flagged, and its latest activity."""
    course: Course
    progress_percentage: int
    flagged: bool
    last_activity: datetime | None


def ranked_course_summaries(
    learner: Learner, details: list[LearnerCourseDetail]
) -> list[RankedCourseSummary]:
    # One summary per course: the highest completion across its registrations,
    # flagged when any registration is, and the latest last_accessed_time across
    # its course progress records (prefetched). Sorted by _rank.


def _rank(summary: RankedCourseSummary) -> tuple[int, bool, float, str]:
    # Flagged courses need the educator first, then the ones under way.
    percentage = summary.progress_percentage
    group = 0 if summary.flagged else 1 if 0 < percentage < 100 else 2
    latest = summary.last_activity
    return (group, latest is None, -latest.timestamp() if latest else 0.0, summary.course.title)


_COURSE_DETAILS_KEY = "_page_course_details"


def attach_course_details(learner: Learner, details: list[LearnerCourseDetail]) -> None:
    vars(learner)[_COURSE_DETAILS_KEY] = details


def attached_course_details(learner: Learner) -> list[LearnerCourseDetail] | None:
    return cast("list[LearnerCourseDetail] | None", vars(learner).get(_COURSE_DETAILS_KEY))


class LearnerDataTable(DataTable):

        cls,
        request: HttpRequest,
        queryset: QuerySet | Sequence[object],
        query: TableQuery,
    ) -> Page:

        learners = cast("list[Learner]", list(page.object_list))
        if learners:

            for learner in learners:
                attach_course_details(learner, details[learner.pk])
        page.object_list = learners
        return page
```

```js
    return this.running ? "This is running and can't be stopped. Wait for it to finish." : "";
},
// Only the confirm of a run that happens in the request locks the dialog;

_isInRequestRun(elt) {

trackListener(this._handlers, document, "htmx:beforeRequest", (event) => {
```
