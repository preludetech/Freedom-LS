### Slice 1: A pause moves every unmet deadline and the time limit when read

1. `shifts.py`, pure (no model imports at runtime; model types under `TYPE_CHECKING`):

   ```python
   SECONDS_PER_DAY = 86_400

   def paused_days(started_at: datetime, end_at: datetime) -> int:
       # Whole days, rounded up: a pause of one second costs the learner a day of
       # calendar, so it gives a day back. Zero only when nothing elapsed.
       seconds = (end_at - started_at).total_seconds()
       return max(0, math.ceil(seconds / SECONDS_PER_DAY))

   @dataclass(frozen=True)
   class ShiftSources:
       pauses: Sequence[CoursePause]
       completed_at: Mapping[UUID, datetime]     # collection item id -> completion instant

   def shift_for_time_limit(sources, now, tz) -> int:
       return sum(p.paused_days(now, tz) for p in sources.pauses)

   def shift_for_collection_item(sources, collection_item_id, now, tz) -> int:
       done = sources.completed_at.get(collection_item_id)
       # A pause that began after the item was met leaves its date alone.
       return sum(
           p.paused_days(now, tz)
           for p in sources.pauses
           if done is None or p.started_at <= done
       )
   ```

   ```python
   def planned_resume_instant(self, tz: ZoneInfo) -> datetime | None:
       # Midnight starting the planned resume day, in the site zone.
       if self.planned_resume_on is None:
           return None
       return datetime.combine(self.planned_resume_on, time.min, tzinfo=tz)

   def end_instant(self, tz: ZoneInfo) -> datetime | None:
       # The earlier of the recorded end and the planned resume instant.
       planned = self.planned_resume_instant(tz)
       ends = [i for i in (self.ended_at, planned) if i is not None]
       return min(ends) if ends else None

   def is_open(self, now, tz) -> bool:
       end = self.end_instant(tz)
       return end is None or end > now

   def paused_days(self, now, tz) -> int:
       end = self.end_instant(tz)
       return shifts.paused_days(self.started_at, min(end, now) if end else now)
   ```

2. `deadlines/actions.py`:

   ```python
   class DeadlineActionError(ValueError):
       """Input a staff member can correct; the admin shows its message on the form."""

   def _locked(record) -> CourseProgress:
       # _base_manager: the sweep calls these with no request site.
       # of=("self",): lock the record alone, not the course, learner and site rows
       # the join reads, which unrelated actions and content loads also touch.
       return CourseProgress._base_manager.select_for_update(of=("self",)).select_related(
           "learner__user", "course", "site").get(pk=record.pk)

   def _check(record, reason) -> str:
       # Returns the reason as it is stored, so the row and the audit entry cannot differ.
       reason = reason.strip()
       if not reason: raise DeadlineActionError("Give a reason.")
       if len(reason) > REASON_LENGTH: raise DeadlineActionError(...)
       if record.completed_time is not None:
           raise DeadlineActionError("This course is complete.")
       return reason

   def _notify(record, actor, category, data) -> None:
       # The person who acted already knows; everyone else was not there.
       if actor is not None and actor.pk == record.learner.user_id:
           return
       raise_notification(user=record.learner.user, category=category,
                          target=record.course, site_id=record.site_id, data=data)

   def pause_course(record, *, actor, reason, planned_resume_on=None, now=None) -> CoursePause:
       now = now or timezone.now()
       with transaction.atomic():
           locked = _locked(record)
           reason = _check(locked, reason)
           tz = site_time_zone(locked.site)
           if planned_resume_on is not None and planned_resume_on <= now.astimezone(tz).date():
               raise DeadlineActionError("Pick a resume date after today.")
           unended = CoursePause._base_manager.filter(
               course_progress=locked, ended_at__isnull=True).first()
           if unended is not None:
               if unended.is_open(now, tz):
                   raise DeadlineActionError("This course is already paused.")
               # Its planned resume passed with nothing written. Record and announce
               # it now, so the learner and integrators hear "resumed" before "paused"
               # and not after.
               _announce_planned_resume(unended, locked, now)
           pause = CoursePause._base_manager.create(site=locked.site, course_progress=locked,
               started_at=now, planned_resume_on=planned_resume_on, reason=reason,
               started_by=actor)
           record_learner_entry(locked.learner, "course.paused", actor, reason=reason,
               subjects=[locked.course],
               details={"course": locked.course.title,
                        "planned_resume_on": iso date or None})
           fire_webhook_event_on_commit("course.paused", {...}, locked.site)
           _notify(locked, actor, "course.paused", {"course_title": ..., "resume_text": ...})
       return pause

   def resume_course(record, *, actor, reason, now=None) -> CoursePause | None:
       # None, writing nothing, when the record is not paused: a repeated post is a no-op.
       # Sets ended_at=now, ended_by, ended_reason and resume_announced_at=now, so the
       # sweep never announces a pause staff already ended.
       # shift_days = pause.paused_days(now, tz), read after ended_at is set.
   ```

   `_announce_planned_resume(pause, locked, now) -> bool` is the private body the sweep's
   entry point (a later slice) also calls, so the two cannot drift. Under the record lock the
   caller already holds:

   ```python
   planned = pause.planned_resume_instant(site_time_zone(locked.site))
   if planned is None or planned > now: return False
   # A conditional update is the claim: of two callers, one row count is 1.
   claimed = CoursePause._base_manager.filter(
       pk=pause.pk, resume_announced_at__isnull=True).update(resume_announced_at=now)
   CoursePause._base_manager.filter(pk=pause.pk, ended_at__isnull=True).update(ended_at=planned)
   if not claimed: return False
   _record_resumed(pause, locked, actor=None, resumed_at=planned, planned=True)
   return True
   ```

   `_record_resumed(pause, locked, *, actor, resumed_at, planned, reason="")` writes the
   `course.resumed` audit entry, defers the webhook and notifies; `resume_course` calls it
   with `planned=False`. The summary is "Your educator resumed ..." for a staff resume and
   "<course> has resumed. ..." when planned.

   Payloads open with `learner_webhook_payload(locked.learner)`; instants are `.isoformat()`;
   `actor_id` is `actor.pk` or `None`; `planned_resume_at` is `pause.planned_resume_instant(tz)`;
   `course.resumed` adds `resumed_at`, `shift_days`, `planned: False`. The notification
   `resume_text` and `summary` strings are built here with `formats.date_format(...,
   DATE_DISPLAY_FORMAT)` and `ngettext` for "1 day" / "N days".
