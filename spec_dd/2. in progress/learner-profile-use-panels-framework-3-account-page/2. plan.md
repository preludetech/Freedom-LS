```python
def account_area(
    request: HttpRequest, path_string: str = ""
) -> HttpResponse | StreamingHttpResponse:
    """Serve the panel framework for the signed-in user's own account."""
    if not path_string.strip("/"):
        landing = first_visible_section(ACCOUNT_AREA_NAV, request)
        if landing is None:
            raise Http404
        return redirect("accounts:account_profile", path_string=landing.url_name)
    # The framework's navigation response swaps in its own one-entry
    # #sidebar-nav; the account nav, appended after it, replaces it again.
    request.panel_extra_oob = ["accounts/partials/sidebar_nav.html"]
    return panel_framework_view(
        ACCOUNT_AREA_NAV,
        request,
        path_string,
        template_name="accounts/account_page.html",
        url_name="accounts:account_profile",
    )
```

```python
def get_own_user(request: HttpRequest) -> User:
    """A fresh copy of the signed-in user.

    A ModelForm writes submitted values onto its instance while it
    validates, and request.user is what the header renders, so a form gets
    its own copy and a refused submission never shows there. The base
    manager, because the lookup is by the signed-in user's own pk, which the
    session already vouches for, and `objects` filters to the request's site.
    """
    return User._base_manager.get(pk=request.user.pk)
```

The docstring folds in today's `_own_user` reasoning. `_own_user` stays in `views.py` until
slice 3 deletes it.

#### Section config: `panels/personal_details/config.py`

```python
class PersonalDetailsConfig(ObjectViewConfig):
    url_name = "personal-details"
    menu_label = "Personal details"
    instance_view = PersonalDetailsInstanceView

    @classmethod
    def get_object(cls, request: HttpRequest) -> Model:
        return get_own_user(request)

    @classmethod
    def authorise_instance(cls, request: HttpRequest, instance: Model) -> None:
        if instance.pk != request.user.pk:
            raise Http404

    @classmethod
    def has_capability(cls, request: HttpRequest, capability: str, scope: Model) -> bool:
        return (
            capability in ACCOUNT_CAPABILITIES
            and isinstance(scope, User)
            and scope.pk == request.user.pk
        )

    @classmethod
    def get_instance_label(cls, instance: Model) -> str:
        return "Personal details"
```

#### Actions: `panels/personal_details/actions.py`

```python
ACCOUNT_CHANGED = "accountChanged"

def account_saved_response(
    ctx: PanelContext,
    action: PanelAction,
    html: str,
    message: str,
    *,
    close_modal: bool = False,
) -> HttpResponse:
    """200, not EditAction's 204: htmx drops a 204's body, and this body
    carries the re-rendered form. The queued message becomes the toast
    through HtmxMessagesMiddleware."""
    messages.success(ctx.request, message)
    response = HttpResponse(html)
    response["HX-Trigger"] = build_hx_trigger(
        action.get_success_events(ctx.instance), close_modal=close_modal
    )
    return response

class AccountEditAction(EditAction):
    """Saves part of the signed-in user's own account from a form in a panel."""

    renders_in_footer = False
    autofocus_first_field = False
    success_message: str

    def __init__(self, user: User, form_class: type[forms.ModelForm], form_title: str) -> None:
        super().__init__(
            form_class=form_class,
            form_title=form_title,
            instance=user,
            success_events=(ACCOUNT_CHANGED,),
        )

    def get_capability(self, ctx: PanelContext) -> str | None:
        # Named explicitly: EditAction falls back to the model's change
        # permission, which a learner does not hold.
        return self.capability

    def form_valid(self, ctx: PanelContext, form: forms.BaseForm) -> HttpResponse:
        cast(forms.ModelForm, form).save()
        html = render_to_string(
            self.get_fragment_template_name(),
            self.get_fragment_context(ctx, self.get_unbound_form(ctx)),
            request=ctx.request,
        )
        return account_saved_response(ctx, self, html, self.success_message)

class EditNameAction(AccountEditAction):
    label = "Edit name"
    action_name = "edit_name"
    capability = CHANGE_OWN_NAME
    template_name = "accounts/partials/name_form.html"
    success_message = "Name saved"

    def __init__(self, user: User) -> None:
        super().__init__(user, form_class=UserForm, form_title="Name")
```

No audit entry is written. `get_unbound_form` re-instantiates with `ctx.instance`, which
`form.save()` just wrote, so the re-rendered form holds the saved names.

### Slice 7: the inline record form has no space under its field row

Behaviour: neither the hidden request-error element nor the base `form` spacing rule adds
space under an inline record form's field row, on educator and account pages alike.

**Builds on:** slices 2, 3, 4, 5
**Touches:** freedom_ls/panel_framework/templates/panel_framework/partials/inline_record_form.html, freedom_ls/tests/playwright_helpers.py, freedom_ls/educator_interface/tests/playwright/test_learner_page_flow.py, freedom_ls/accounts/tests/playwright/test_profile_picture_flow.py

The cause: `space-y-3` on the form gives every child but the last a bottom margin. The hidden
`[data-request-error]` div is now the last child, so the field row gets 12px. The base
`form { @apply space-y-2 }` rule in `tailwind.components.css` does the same.

`flex flex-col gap-3 space-y-0`, as `profile_picture_form.html` already has. A flex gap is only
laid out between children that generate a box, so the hidden element adds nothing. `space-y-0`
is a utility, so it beats the base component rule and cancels its sibling margins on this form.

the form uses `gap` and not `space-y`. A flex column stretches its children's width: check that
