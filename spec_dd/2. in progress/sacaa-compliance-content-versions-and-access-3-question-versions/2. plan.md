## Decisions

- **Raw SQL, reviewed.** `delete_question_versions()` and migration `0021` use raw SQL because the
  queryset guard and the trigger both refuse an ORM delete. The table name comes from
  `connection.ops.quote_name(FormQuestionVersion._meta.db_table)`, the trigger name is a module
  constant, the ids are a bound `%s::uuid[]` parameter, and `RunSQL` is a fixed string. Nothing
  user-supplied reaches the SQL, and the delete runs only from development commands.
- **`site_id` is set by hand from the question.** A management command and a migration have no
  request to fill it from, and a version's site must be its question's. A test asserts that.
- **Version loads use `FormQuestionVersion.unfiltered`**, the guarded manager that does not
  filter by site, which `Meta.base_manager_name` names. Readers such as the applications API serve
  requests whose host is not the row's site (`course_applications/api.py` `answers_prefetch`), and
  the ids they look up come from site-scoped rows already.
- **Unsaved version-backed instances never reach a related filter.** Every `question=question`
  filter that can receive a path question becomes `question_id=question.id`:
  `form_engine/models.py` (`save_answers` x2, `score_category_value_sum`, `get_incorrect_quiz_answers`),
  `form_engine/paging.py` `is_answered`, `form_engine/views.py` (upload view x2),
  `learner_interface/views.py` (around line 2787). Each slice changes the ones it first feeds a
  version-backed question.
- **A quiz file upload records the sitting's pinned version.** The upload view takes the question
  from the sitting's path, which is version-backed for a pinned quiz question, the same rule
  `save_answers()` follows.
- **Admin deletion of a versioned form is refused by `PROTECT`.** Deleting a form whose questions
  have versions fails until the admin question pages go. Only development commands delete
  versions.
- **Plan-level names** the spec does not name, all private or local: `version_options()` (the
  canonical option list builder), `FormQuestionVersion.option_rows()`, `SavedQuestionVersion`
  (what `save_question_version` returns), `_RunLedger.question_versions`, `TRIGGER_NAME`, the
  `unfiltered` manager.

## Shape of the change

- `FormQuestionVersion` and its helpers live in `form_engine/models.py`. `form_engine` never
  imports `content_engine`, in code or tests.
- `FormQuestion` and `QuestionOption` gain a class attribute `pinned_version: FormQuestionVersion
  | None = None` (not a field) and `save()`/`delete()` overrides that raise when it is set.
- `served_path()` is the one place the quiz runner, scoring, results page and educator review get
  version-backed questions.
- Read-only readers of an answer (admin answer views, export, API, check-your-answers) go through
  `answered_questions()` and `answered_options()`.

class QuestionVersionOption(BaseModel):
    """One option as a version holds it."""

    model_config = ConfigDict(extra="forbid")
    id: str
    text: str
    value: str
    order: int
    correct: bool | None


def version_options(options: Iterable[QuestionOption]) -> list[dict[str, object]]:
    """The options as a version stores them: ordered by order then id, validated."""
    rows = sorted(options, key=lambda o: (o.order, str(o.id)))
    return [
        QuestionVersionOption(
            id=str(o.id), text=o.text, value=str(o.value), order=o.order, correct=o.correct
        ).model_dump()
        for o in rows
    ]


_IMMUTABLE_MESSAGE = "FormQuestionVersion is immutable: a saved version cannot change"


class FormQuestionVersionQuerySet(models.QuerySet["FormQuestionVersion"]):
    def update(self, **kwargs: object) -> int:
        raise ValueError(_IMMUTABLE_MESSAGE)

    def bulk_update(self, *args: object, **kwargs: object) -> int:
        raise ValueError(_IMMUTABLE_MESSAGE)

    def delete(self) -> tuple[int, dict[str, int]]:
        raise ValueError(_IMMUTABLE_MESSAGE)


class FormQuestionVersion(SiteAwareModel):
    question = models.ForeignKey(FormQuestion, on_delete=models.PROTECT, related_name="versions")
    number = models.PositiveIntegerField()
    question_text = models.TextField()
    explanation = models.TextField(blank=True, default="")
    type = models.CharField(max_length=20, choices=QuestionType.choices)
    required = models.BooleanField(default=True)
    min = models.CharField(max_length=20, blank=True, default="")
    max = models.CharField(max_length=20, blank=True, default="")
    decimal_places = models.PositiveSmallIntegerField(default=0)
    options = models.JSONField(default=list)
    saved_at = models.DateTimeField(auto_now_add=True)
    source_commit = models.CharField(max_length=64, blank=True)
    from_migration = models.BooleanField(default=False)

    objects = SiteAwareManager.from_queryset(FormQuestionVersionQuerySet)()
    # Serves related lookups such as answer.question_version, which must not
    # depend on the site of the request that makes them.
    unfiltered = models.Manager.from_queryset(FormQuestionVersionQuerySet)()

    class Meta:

        constraints = [
            models.UniqueConstraint(fields=["question", "number"], name="one_number_per_question_version")

`save_question_version`:

```python
@dataclass(frozen=True)
class SavedQuestionVersion:
    version: FormQuestionVersion | None  # None when nothing differed
    previous_number: int | None
    changed: list[str]


def save_question_version(
    question: FormQuestion, options: Sequence[QuestionOption], source_commit: str
) -> SavedQuestionVersion:
    with transaction.atomic():
        # Locks the live row so two saves of one question cannot take the same number.
        FormQuestion._base_manager.select_for_update().get(pk=question.pk)
        latest = FormQuestionVersion.unfiltered.filter(question_id=question.pk).order_by("-number").first()
        changed = latest.differs_from(question, options) if latest else []
        if latest is not None and not changed:
            return SavedQuestionVersion(None, latest.number, [])
        version = FormQuestionVersion.objects.create(
            site_id=question.site_id, question=question,
            number=latest.number + 1 if latest else 1,
            question_text=question.question, type=question.type, required=question.required,
            min=question.min, max=question.max, decimal_places=question.decimal_places,
            explanation=question.explanation, options=version_options(options),
            source_commit=source_commit,
        )
        return SavedQuestionVersion(version, latest.number if latest else None, changed)
```

def delete_question_versions(questions: Iterable[FormQuestion]) -> None:
    """Remove every version of these questions. For development commands only; callers
    delete the answers pinned to the versions first, or the PROTECT foreign key refuses."""
    ids = [str(question.pk) for question in questions]
    if not ids:
        return
    table = connection.ops.quote_name(FormQuestionVersion._meta.db_table)
    with transaction.atomic(), connection.cursor() as cursor:
        # The queryset guard refuses an ORM delete.
        cursor.execute(f"DELETE FROM {table} WHERE question_id = ANY(%s::uuid[])", [ids])

      # A shallow copy keeps every non-versioned field and the cached form_page and
      # condition_option, so building one costs no query.
      question = copy.copy(live)
      question.__dict__.pop("_prefetched_objects_cache", None)
      question.question = self.question_text
      question.type, question.required = self.type, self.required
      question.min, question.max = self.min, self.max

```python
def answered_questions(answers: Iterable[QuestionAnswer]) -> dict[UUID, FormQuestion]:
    """The question each of one sitting's answers was given on: the version-backed
    question for an answer that records a version, else the live question. Expects
    each answer's question to be loaded; a version already loaded costs no query."""
    answers = list(answers)
    loaded = {a.question_version_id: a.question_version for a in answers
              if a.question_version_id and QuestionAnswer.question_version.is_cached(a)}
    wanted = {a.question_version_id for a in answers if a.question_version_id} - loaded.keys()
    versions = loaded | (FormQuestionVersion.unfiltered.in_bulk(wanted) if wanted else {})
    return {
        a.question_id: versions[a.question_version_id].as_question(a.question)
        if a.question_version_id else a.question
        for a in answers
    }


def answered_options(answer: QuestionAnswer, question: FormQuestion) -> list[QuestionOption]:
    selected = list(answer.selected_options.all())
    if question.pinned_version is not None:
        held = {option.id: option for option in question.options.all()}
        selected = [held.get(option.id, option) for option in selected]
    return sorted(selected, key=lambda option: option.order)
```
